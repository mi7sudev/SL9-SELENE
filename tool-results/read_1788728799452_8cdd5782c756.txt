     1→// mcp-connections.cjs — MCP connection lifecycle (the "control plane") for
     2→// lumo-server.cjs.
     3→//
     4→// A connection is a user/tenant-scoped, credentialed binding to an
     5→// admin-configured MCP server definition. This module owns:
     6→//   - connection records in the SQLite store (store.cjs; credentials
     7→//     AES-256-GCM encrypted inside the record)
     8→//   - data/secret.key            (AES-256-GCM key, created on first use)
     9→//   - the lifecycle state machine (authorizing → connected → ready |
    10→//     failed | needs_reauth, plus revoked)
    11→//   - one-time OAuth state flows (state + PKCE) and API-key setup flows
    12→//   - the agent control-plane tools (lumo__*) and their system note
    13→//
    14→// Security invariants (see MCP.md):
    15→//   - Credentials only ever live here, encrypted at rest; everything that
    16→//     leaves this module is redacted status. Control-plane tool results,
    17→//     catalog views, and logs never contain a credential value.
    18→//   - Chat can never create or modify server definitions: connections bind
    19→//     ONLY to admin-registered definitions resolved via getServerDef.
    20→//   - One-time flows: every flow token/state is single-use and expiring.
    21→//
    22→// State machine:
    23→//   available (implicit: no record) → authorizing → connected → ready
    24→//       authorizing/connected → failed | needs_reauth
    25→//       ready → (revoke) → revoked;   failed/needs_reauth → authorizing (retry)

   104→function b64url(buf) { return Buffer.from(buf).toString('base64url'); }
   105→function s256(verifier) { return b64url(crypto.createHash('sha256').update(verifier).digest()); }
   106→function randomToken(bytes = 24) { return crypto.randomBytes(bytes).toString('hex'); }

   108→// `store` is the SQLite store handle from store.cjs (the mcp_connections
   109→// table); `dataDir` still locates secret.key, which deliberately stays a
   110→// plain 0600 file so the encryption key never lives beside the ciphertext.
   111→function createConnectionService({ dataDir, log, nowIso, store, getServerDef, mcpManager }) {
   112→    const SECRET_KEY_FILE = path.join(dataDir, 'secret.key');
   113→
   114→    const say = (...a) => { try { log && log(...a); } catch { /* logging never throws */ } };
   115→
   116→    // ── credential encryption (AES-256-GCM, key material in data/secret.key) ──
   117→    let secretKey = null;
   118→    function ensureSecretKey() {
   119→        if (secretKey) return secretKey;
   120→        try {
   121→            const buf = fs.readFileSync(SECRET_KEY_FILE);
   122→            if (buf.length === 32) { secretKey = buf; return secretKey; }
   123→        } catch { /* first use */ }
   124→        secretKey = crypto.randomBytes(32);
   125→        try { fs.writeFileSync(SECRET_KEY_FILE, secretKey, { mode: 0o600 }); } catch (e) {
   126→            say(`[MCP-CONN] WARNING: could not persist secret key: ${e && e.message}`);
   127→        }
   128→        return secretKey;
   129→    }
   130→    function encryptCredential(plainText) {
   131→        const key = ensureSecretKey();
   132→        const iv = crypto.randomBytes(12);
   133→        const cipher = crypto.createCipheriv('aes-256-gcm', key, iv);
   134→        const ct = Buffer.concat([cipher.update(String(plainText), 'utf8'), cipher.final()]);
   135→        const tag = cipher.getAuthTag();
   136→        return `v1:${iv.toString('base64')}:${tag.toString('base64')}:${ct.toString('base64')}`;
   137→    }
   138→    function decryptCredential(blob) {
   139→        if (typeof blob !== 'string' || !blob.startsWith('v1:')) return null;
   140→        const [, ivB64, tagB64, ctB64] = blob.split(':');
   141→        try {
   142→            const decipher = crypto.createDecipheriv('aes-256-gcm', ensureSecretKey(), Buffer.from(ivB64, 'base64'));
   143→            decipher.setAuthTag(Buffer.from(tagB64, 'base64'));
   144→            return Buffer.concat([decipher.update(Buffer.from(ctB64, 'base64')), decipher.final()]).toString('utf8');
   145→        } catch {
   146→            return null; // tampered or wrong key -> treat as no credential
   147→        }
   148→    }

   182→    function updateRecord(id, patch) {
   183→        const rec = store.getConnectionRecord(id);
   184→        if (!rec) return null;
   185→        Object.assign(rec, patch, { updatedAt: nowIso() });
   186→        store.upsertConnectionRecord(rec);

   207→    // ── one-time flows (in-memory; they die with the process by design) ──────
   208→    const oauthFlows = new Map();  // state -> { serverId, ownerUid, connectionId, verifier, createdAt, used }
   209→    const setupFlows = new Map();  // flowId -> { serverId, ownerUid, connectionId, createdAt, used }
   210→
   211→    function sweepFlows() {
   212→        const now = Date.now();
   213→        for (const [k, f] of oauthFlows) if (f.used || now - f.createdAt > OAUTH_STATE_TTL_MS) oauthFlows.delete(k);
   214→        for (const [k, f] of setupFlows) if (f.used || now - f.createdAt > SETUP_FLOW_TTL_MS) setupFlows.delete(k);
   215→    }

   217→    // ── status derivation ─────────────────────────────────────────────────────
   218→    // Maps a live transport state to the honest user-facing catalog status.
   219→    function mapTransportStatus(st) {
   220→        if (st === 'connected') return 'ready';
   221→        if (st === 'error') return 'unavailable';
   222→        return 'not_discovered'; // disconnected / connecting: nothing discovered yet
   223→    }

   225→    // Full user-facing view of one server definition (for catalog + control
   226→    // tools). Tools are names + classification only — never schemas/env.
   227→    function connectionStatusFor(def, uid) {
   228→        const entry = def; // raw definition; manager normalizes
   229→        if (!entry.enabled) return { serverId: entry.id, name: entry.name, auth: entry.auth || 'none', status: 'disabled', tools: null, lastDiscoveredAt: null };
   230→        if (!entry.auth || entry.auth === 'none') {
   231→            const st = mcpManager.statusView(entry);
   232→            return {
   233→                serverId: entry.id,
   234→                name: entry.name,
   235→                auth: 'none',
   236→                status: mapTransportStatus(st.status),
   237→                tools: st.status === 'connected' ? st.tools.map((t) => ({ name: t.name, classification: t.classification })) : null,
   238→                lastDiscoveredAt: st.lastDiscoveredAt,
   239→            };
   240→        }
   241→        const rec = findConnection(entry.id, uid);
   242→        if (!rec) return { serverId: entry.id, name: entry.name, auth: entry.auth, status: 'available', tools: null, lastDiscoveredAt: null };
   243→        let status = rec.status;
   244→        let tools = null;
   245→        if (rec.status === 'ready') {
   246→            const st = mcpManager.statusView(entry, { connection: { id: rec.id } });
   247→            if (st.status === 'error') status = 'unavailable';
   248→            else tools = st.tools.map((t) => ({ name: t.name, classification: t.classification }));
   249→        }
   250→        return {
   251→            serverId: entry.id,
   252→            name: entry.name,
   253→            auth: entry.auth,
   254→            status,
   255→            error: rec.error || null,
   256→            tools,
   257→            lastValidatedAt: rec.lastValidatedAt || null,
   258→            lastDiscoveredAt: rec.lastDiscoveredAt || null,
   259→        };
   260→    }

   262→    // The decrypted credential value for a ready connection (transport header
   263→    // injection). Returns { id, value } or null. Value never leaves this
   264→    // module except into the manager's transport builder / redaction set.
   265→    function findReadyCredential(serverId, uid) {
   266→        const rec = findConnection(serverId, uid);
   267→        if (!rec || rec.status !== 'ready' || !rec.credential) return null;
   268→        const raw = decryptCredential(rec.credential);
   269→        if (!raw) return null;
   270→        let parsed;
   271→        try { parsed = JSON.parse(raw); } catch { return null; }
   272→        if (!parsed || typeof parsed.value !== 'string' || !parsed.value) return null;
   273→        return { id: rec.id, value: parsed.value };
   274→    }

   276→    // ── validation: a connection is ready only after a REAL connect + discovery
   277→    async function validateConnection(rec, def) {
   278→        let credential = null;
   279→        try { credential = JSON.parse(decryptCredential(rec.credential) || 'null'); } catch { credential = null; }
   280→        const credValue = credential && typeof credential.value === 'string' ? credential.value : '';
   281→        updateRecord(rec.id, { status: 'connected', error: null });
   282→        try {
   283→            await mcpManager.connect(def, { force: true, connection: { id: rec.id, credential: credValue } });
   284→            const st = mcpManager.statusView(def, { connection: { id: rec.id } });
   285→            // ready only after a REAL connect + tool discovery: the manager
   286→            // stamps lastDiscoveredAt solely when tools/list actually answered,
   287→            // so a transport that connects but cannot be enumerated never
   288→            // reaches ready (no fake success)
   289→            if (st.status === 'connected' && st.lastDiscoveredAt) {
   290→                updateRecord(rec.id, {
   291→                    status: 'ready',
   292→                    error: null,
   293→                    lastValidatedAt: nowIso(),
   294→                    lastDiscoveredAt: st.lastDiscoveredAt,
   295→                });
   296→                say(`[MCP-CONN] connection ready: ${def.name} (${def.id}) owner=${rec.ownerUid} tools=${st.tools.length}`);
   297→            } else {
   298→                const code = st.status === 'connected' ? 'discovery_failed' : 'validation_failed';
   299→                updateRecord(rec.id, { status: 'failed', error: { code } });
   300→                say(`[MCP-CONN] connection validation failed: ${def.name} (${def.id}): ${code}`);
   301→            }
   302→        } catch (e) {
   303→            const msg = String((e && e.message) || '');
   304→            const code = /401|403|unauthorized|invalid[_ ]?token|bad credentials|invalid api key/i.test(msg)
   305→                ? 'auth_rejected' : 'validation_failed';
   306→            updateRecord(rec.id, { status: code === 'auth_rejected' ? 'needs_reauth' : 'failed', error: { code } });
   307→            say(`[MCP-CONN] connection validation failed: ${def.name} (${def.id}): ${mcpManager.redact(msg)}`);
   308→        }
   309→        return findConnectionById(rec.id);
   310→    }

   312→    // ── lifecycle operations (used by control tools AND HTTP endpoints) ──────
   313→    async function startConnect({ serverId, uid, confirm, baseUrl }) {
   314→        sweepFlows();
   315→        // per-uid cap on live authorization flows: the TTL sweep bounds the map
   316→        // overall, this bounds how many one account can have open at once
   317→        let liveFlows = 0;
   318→        for (const f of oauthFlows.values()) if (f.ownerUid === uid) liveFlows++;
   319→        for (const f of setupFlows.values()) if (f.ownerUid === uid) liveFlows++;
   320→        if (liveFlows >= 20) {
   321→            return { ok: false, error: 'too_many_flows', message: 'Too many pending authorizations for this account. Finish or let one expire, then try again.' };
   322→        }
   323→        const def = getServerDef(String(serverId || ''));
   324→        if (!def) return { ok: false, error: 'unknown_connection', message: 'No MCP connection with that id is configured on this instance. The administrator must add it first.' };
   325→        if (!def.enabled) return { ok: false, error: 'disabled', message: `"${def.name}" is disabled by the administrator.` };
   326→        const auth = def.auth || 'none';
   327→        if (auth === 'none') {
   328→            return { ok: true, serverId: def.id, name: def.name, status: 'ready', message: `"${def.name}" is instance-shared (administrator-managed credentials) — its tools are already available in chats; no personal connection is needed.` };
   329→        }
   330→        if (confirm !== true) {
   331→            return { ok: false, needsConfirmation: true, serverId: def.id, name: def.name, message: `Ask the user to confirm connecting to "${def.name}" before proceeding.` };
   332→        }
   333→        let rec = findConnection(def.id, uid);
   334→        if (rec && rec.status === 'ready' && findReadyCredential(def.id, uid)) {
   335→            return { ok: true, serverId: def.id, name: def.name, status: 'ready', message: `"${def.name}" is already connected.` };
   336→        }
   337→        if (!rec) rec = createRecord(def.id, uid, 'authorizing');
   338→        else updateRecord(rec.id, { status: 'authorizing', error: null });
   339→
   340→        if (auth === 'api_key') {
   341→            const flowId = randomToken(24);
   342→            setupFlows.set(flowId, { serverId: def.id, ownerUid: uid, connectionId: rec.id, createdAt: Date.now(), used: false });
   343→            const setupUrl = `${baseUrl}/mcp/setup/${flowId}`;
   344→            say(`[MCP-CONN] api-key flow started: ${def.id} owner=${uid} flow=${flowId.slice(0, 8)}…`);
   345→            return { ok: true, serverId: def.id, name: def.name, status: 'authorizing', setupUrl, message: `Give this URL to the user: they must open it in a browser and enter their API key there. Never ask for the key in chat. Then verify with connection_status before claiming success.` };
   346→        }
   347→        // oauth
   348→        const cfg = def.oauth || {};
   349→        if (!cfg.authorizeUrl || !cfg.tokenUrl || !cfg.clientId) {
   350→            updateRecord(rec.id, { status: 'failed', error: { code: 'misconfigured' } });
   351→            return { ok: false, error: 'misconfigured', message: `"${def.name}" has an incomplete OAuth configuration; the administrator must fix it.` };
   352→        }
   353→        const state = randomToken(24);
   354→        const verifier = b64url(crypto.randomBytes(32));
   355→        oauthFlows.set(state, { serverId: def.id, ownerUid: uid, connectionId: rec.id, verifier, createdAt: Date.now(), used: false });
   356→        const u = new URL(cfg.authorizeUrl);
   357→        u.searchParams.set('response_type', 'code');
   358→        u.searchParams.set('client_id', cfg.clientId);
   359→        u.searchParams.set('redirect_uri', `${baseUrl}/mcp/oauth/callback`);
   360→        u.searchParams.set('state', state);
   361→        if (cfg.scopes) u.searchParams.set('scope', cfg.scopes);
   362→        if (cfg.usePkce !== false) {
   363→            u.searchParams.set('code_challenge', s256(verifier));
   364→            u.searchParams.set('code_challenge_method', 'S256');
   365→        }
   366→        say(`[MCP-CONN] oauth flow started: ${def.id} owner=${uid} state=${state.slice(0, 8)}…`);
   367→        return { ok: true, serverId: def.id, name: def.name, status: 'authorizing', authorizeUrl: u.toString(), message: `Give this authorization URL to the user: they must open it, approve access, and they will land back on this instance. Then verify with connection_status before claiming success.` };
   368→    }

   370→    async function completeOAuth({ state, code, error, baseUrl }) {
   371→        sweepFlows();
   372→        if (!state) return { ok: false, page: 'error', code: 'invalid_state' };
   373→        const flow = oauthFlows.get(state);
   374→        if (!flow || flow.used) return { ok: false, page: 'error', code: 'invalid_state' };
   375→        if (error || !code) {
   376→            // the user denied (or abandoned) the authorization at the provider:
   377→            // consume the one-time state, record the honest recoverable state,
   378→            // and never store a credential
   379→            oauthFlows.delete(state);
   380→            const rec = findConnectionById(flow.connectionId);
   381→            if (rec && rec.ownerUid === flow.ownerUid) {
   382→                updateRecord(rec.id, { status: 'failed', error: { code: 'auth_denied' } });
   383→            }
   384→            say(`[MCP-CONN] oauth ${error ? `denied (${String(error).slice(0, 40)})` : 'abandoned'}: ${flow.serverId} owner=${flow.ownerUid}`);
   385→            const def = getServerDef(flow.serverId);
   386→            return { ok: false, page: 'denied', code: 'auth_denied', name: def ? def.name : undefined };
   387→        }
   388→        oauthFlows.delete(state); // one-time use, even before validation succeeds
   389→        const def = getServerDef(flow.serverId);
   390→        if (!def) return { ok: false, page: 'error', code: 'unknown_connection' };
   391→        const rec = findConnectionById(flow.connectionId);
   392→        if (!rec || rec.ownerUid !== flow.ownerUid) return { ok: false, page: 'error', code: 'invalid_state' };
   393→        const cfg = def.oauth || {};
   394→        try {
   395→            const form = new URLSearchParams();
   396→            form.set('grant_type', 'authorization_code');
   397→            form.set('code', String(code));
   398→            form.set('redirect_uri', `${baseUrl}/mcp/oauth/callback`);
   399→            form.set('client_id', cfg.clientId);
   400→            if (cfg.clientSecret) form.set('client_secret', cfg.clientSecret);
   401→            if (cfg.usePkce !== false) form.set('code_verifier', flow.verifier);
   402→            const resp = await fetch(cfg.tokenUrl, {
   403→                method: 'POST',
   404→                headers: { 'content-type': 'application/x-www-form-urlencoded', accept: 'application/json' },
   405→                body: form.toString(),
   406→                signal: AbortSignal.timeout(15000),
   407→            });
   408→            if (!resp.ok) throw new Error(`token endpoint ${resp.status}`);
   409→            const tok = await resp.json();
   410→            if (!tok || typeof tok.access_token !== 'string' || !tok.access_token) throw new Error('no access_token in token response');
   411→            const credential = { type: 'oauth', value: tok.access_token, refreshToken: typeof tok.refresh_token === 'string' ? tok.refresh_token : null };
   412→            updateRecord(rec.id, { credential: encryptCredential(JSON.stringify(credential)) });
   413→            mcpManager.registerSecretValue(tok.access_token);
   414→            const finalRec = await validateConnection(findConnectionById(rec.id), def);
   415→            return { ok: finalRec.status === 'ready', page: finalRec.status === 'ready' ? 'ready' : 'failed', status: finalRec.status, name: def.name, error: finalRec.error };
   416→        } catch (e) {
   417→            updateRecord(rec.id, { status: 'failed', error: { code: 'oauth_exchange_failed' } });
   418→            say(`[MCP-CONN] oauth exchange failed: ${def.id}: ${mcpManager.redact(String((e && e.message) || e))}`);
   419→            return { ok: false, page: 'failed', code: 'oauth_exchange_failed', name: def.name };
   420→        }
   421→    }

   423→    async function completeSetup({ flowId, keyValue }) {
   424→        sweepFlows();
   425→        const flow = setupFlows.get(String(flowId || ''));
   426→        if (!flow || flow.used) return { ok: false, page: 'error', code: 'invalid_flow' };
   427→        const key = String(keyValue || '').trim();
   428→        if (!key) return { ok: false, page: 'form', code: 'missing_key' };
   429→        setupFlows.delete(flowId); // one-time use
   430→        const def = getServerDef(flow.serverId);
   431→        if (!def) return { ok: false, page: 'error', code: 'unknown_connection' };
   432→        const rec = findConnectionById(flow.connectionId);
   433→        if (!rec || rec.ownerUid !== flow.ownerUid) return { ok: false, page: 'error', code: 'invalid_flow' };
   434→        const credential = { type: 'api_key', value: key };
   435→        updateRecord(rec.id, { credential: encryptCredential(JSON.stringify(credential)) });
   436→        mcpManager.registerSecretValue(key);
   437→        const finalRec = await validateConnection(findConnectionById(rec.id), def);
   438→        return { ok: finalRec.status === 'ready', page: finalRec.status === 'ready' ? 'ready' : 'failed', status: finalRec.status, name: def.name, error: finalRec.error };
   439→    }

   441→    async function disconnectConnection({ serverId, uid, confirm }) {
   442→        const def = getServerDef(String(serverId || ''));
   443→        if (!def) return { ok: false, error: 'unknown_connection', message: 'No MCP connection with that id is configured on this instance.' };
   444→        if (confirm !== true) {
   445→            return { ok: false, needsConfirmation: true, serverId: def.id, name: def.name, message: `Ask the user to confirm disconnecting "${def.name}" before proceeding. This deletes the stored credential.` };
   446→        }
   447→        const rec = findConnection(def.id, uid);
   448→        if (!rec || rec.status === 'revoked') {
   449→            return { ok: true, serverId: def.id, name: def.name, status: 'available', message: `"${def.name}" had no active user connection.` };
   450→        }
   451→        updateRecord(rec.id, { status: 'revoked', credential: null, error: null });
   452→        try { await mcpManager.disconnect(transportKeyFor(def.id, rec.id), { silent: true }); } catch { /* already dead */ }
   453→        say(`[MCP-CONN] connection revoked: ${def.id} owner=${rec.ownerUid}`);
   454→        return { ok: true, serverId: def.id, name: def.name, status: 'revoked', message: `"${def.name}" is disconnected; the stored credential was deleted.` };
   455→    }

   457→    // Admin override: revoke any record by id (no ownership check — admin only
   458→    // route in lumo-server.cjs). Credential wiped, transport closed.
   459→    async function revokeById(id) {
   460→        const rec = findConnectionById(String(id || ''));
   461→        if (!rec) return null;
   462→        updateRecord(rec.id, { status: 'revoked', credential: null, error: null });
   463→        try { await mcpManager.disconnect(transportKeyFor(rec.serverId, rec.id), { silent: true }); } catch { /* already dead */ }
   464→        say(`[MCP-CONN] connection revoked by admin: ${rec.serverId} owner=${rec.ownerUid}`);
   465→        return findConnectionById(rec.id);
   466→    }

   493→    // Returns { isError, text } for the chat loop. Text is JSON; it never
   494→    // contains credential material.
   495→    async function executeControlTool({ name, args, uid, baseUrl }) {
   496→        const a = args && typeof args === 'object' ? args : {};
   497→        try {
   498→            if (name === `${CONTROL_PREFIX}connections_list`) {
   499→                return { isError: false, text: JSON.stringify(connectionsListFor(uid)) };
   500→            }
   501→            if (name === `${CONTROL_PREFIX}connection_connect`) {
   502→                const r = await startConnect({ serverId: a.serverId, uid, confirm: a.confirm === true, baseUrl });
   503→                return { isError: r.ok === false && !r.needsConfirmation, text: JSON.stringify(r) };
   504→            }
   505→            if (name === `${CONTROL_PREFIX}connection_status`) {
   506→                const def = getServerDef(String(a.serverId || ''));
   507→                if (!def) return { isError: true, text: JSON.stringify({ ok: false, error: 'unknown_connection' }) };
   508→                const v = connectionStatusFor(def, uid);
   509→                return { isError: false, text: JSON.stringify(v) };
   510→            }
   511→            if (name === `${CONTROL_PREFIX}connection_disconnect`) {
   512→                const r = await disconnectConnection({ serverId: a.serverId, uid, confirm: a.confirm === true });
   513→                return { isError: r.ok === false && !r.needsConfirmation, text: JSON.stringify(r) };
   514→            }
   515→            return { isError: true, text: JSON.stringify({ ok: false, error: 'unknown_control_tool' }) };
   516→        } catch (e) {
   517→            say(`[MCP-CONN] control tool ${name} failed:`, mcpManager.redact(String((e && e.message) || e)));
   518→            return { isError: true, text: JSON.stringify({ ok: false, error: 'internal_error' }) };
   519→        }
   520→    }
