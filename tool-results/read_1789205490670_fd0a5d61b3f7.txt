   178→    private async callPostJson(
   179→        resourceName: ResourceName,
   180→        resource: PostableResource,
   181→        path?: string
   182→    ): Promise<RemoteId> {
   183→        const response = await this.sendPostJson(resourceName, resource, path);
   184→        const json = await response.json();
   185→        if (!response.ok) {
   186→            if (response.status === 409) {
   187→                throw new ConflictClientError(`Error during POST ${resourceName}: resource already exists`);
   188→            } else if (response.status === 422 && isLimitReachableResource(resourceName)) {
   189→                // The backend returns 422 when a per-resource limit is reached (e.g.
   190→                // MAX_MESSAGES_PER_CONVERSATION, MAX_ASSETS_PER_SPACE, etc.).
   191→                const code = isProtonApiResponse(json) ? json.Code : undefined;
   192→                const serverMessage =
   193→                    (json && typeof (json as any).Error === 'string' && (json as any).Error) || undefined;
   194→                throw new LimitReachedError(resourceName, { serverMessage, code });
   195→            } else if (response.status >= 400 && response.status < 500) {
   196→                // 4xx don't retry
   197→                throw new ClientError(`Error during POST ${resourceName}: got ${response.status}`);
   198→            }
   199→            throw new Error(`Error during POST ${resourceName}: got ${response.status}`);
   200→        }
   201→        const idExtractor = idExtractorMap[resourceName];
   202→        if (!idExtractor) {
   203→            throw new Error('Unknown resource name');
   204→        }
   205→
   206→        const remoteId = idExtractor(json);
   207→        if (!remoteId) {
   208→            throw new Error("POST: can't find remote ID in response");
   209→        }
   210→        if (!isRemoteId(remoteId)) {
   211→            throw new Error('POST: invalid remote ID in response');
   212→        }
   213→
   214→        return remoteId;
   215→    }

   217→    private async sendPostJson(
   218→        resourceName: ResourceName,
   219→        resource: PostableResource,
   220→        path?: string
   221→    ): Promise<Response> {
   222→        path = !path ? `/${resourceName}` : path.startsWith('/') ? path : `/${path}`;
   223→        const url = `/api/lumo/v1${path}`;
   224→        console.log(`lumo api: http post ${url} <-`, resource);
   225→        const response = await fetch(url, {
   226→            method: 'POST',
   227→            headers: {
   228→                'Content-Type': 'application/json',
   229→                ...this.protonHeaders(),
   230→            },
   231→            body: JSON.stringify(resource),
   232→        });
   233→
   234→        if (response.ok) {
   235→            return response;
   236→        } else {
   237→            if (response.status >= 400 && response.status < 500) {
   238→                return response;
   239→            }
   240→            throw new Error(`Error during POST ${resourceName}: got ${response.status}`);
   241→        }
   242→    }

   400→    // BE response dependent on FF
   401→    public async getMasterKey(): Promise<{ eligibility: number; key: Base64 | null }> {
   402→        const data = await this.callListJson('masterkeys');
   403→        const eligibility = data.Eligibility;
   404→
   405→        if (eligibility !== LUMO_ELIGIBILITY.Eligible) {
   406→            return { eligibility, key: null };
   407→        }
   408→
   409→        const keys = convertMasterKeysFromApi(data.MasterKeys);
   410→
   411→        if (keys.length === 0) {
   412→            return { eligibility, key: null };
   413→        }
   414→
   415→        const key = this.findBestKey(keys);
   416→        return { eligibility, key: key.masterKey };
   417→    }
   418→
   419→    private findBestKey(keys: []): undefined;
   420→    private findBestKey(keys: MasterKey[]): MasterKey;
   421→    private findBestKey(keys: MasterKey[]): MasterKey | undefined {
   422→        return keys.reduce((best: MasterKey | undefined, current: MasterKey) => {
   423→            if (!best) return current;
   424→            if (current.isLatest && !best.isLatest) {
   425→                return current;
   426→            } else if (!current.isLatest && best.isLatest) {
   427→                return best;
   428→            } else if (current.version > best.version) {
   429→                return current;
   430→            } else if (current.version < best.version) {
   431→                return best;
   432→            } else {
   433→                // Compare by createdAt if isLatest and version are the same
   434→                const currentCreatedAt = new Date(current.createdAt);
   435→                const bestCreatedAt = new Date(best.createdAt);
   436→                if (currentCreatedAt > bestCreatedAt) {
   437→                    return current;
   438→                } else if (currentCreatedAt < bestCreatedAt) {
   439→                    return best;
   440→                } else {
   441→                    // In case createdAt is the same(!) we need to make a choice,
   442→                    // let's take the lowest id
   443→                    if (current.id < best.id) {
   444→                        return current;
   445→                    } else {
   446→                        return best;
   447→                    }
   448→                }
   449→            }
   450→        }, undefined);
   451→    }

   560→    public async listSpacesPaginate(): Promise<ListSpacesRemote> {
   561→        let result: ListSpacesRemote = {
   562→            spaces: {},
   563→            conversations: {},
   564→            deletedSpaces: {},
   565→            deletedConversations: {},
   566→            assets: {},
   567→            deletedAssets: {},
   568→        };
   569→        let lastTimestamp: Date | undefined;
   570→        while (true) {
   571→            const createTimeUntil = lastTimestamp ? dateToUnixTimestamp(lastTimestamp) : undefined;
   572→            const next = await this.listSpaces({ createTimeUntil });
   573→            const newLastTimestamp = listify({ ...next.spaces, ...next.deletedSpaces })
   574→                .map((s) => new Date(s.createdAt))
   575→                .reduce(oldestDateReducer, lastTimestamp);
   576→            if (!newLastTimestamp || (lastTimestamp && newLastTimestamp >= lastTimestamp)) break;
   577→            lastTimestamp = newLastTimestamp;
   578→            result = {
   579→                spaces: { ...result.spaces, ...next.spaces },
   580→                conversations: { ...result.conversations, ...next.conversations },
   581→                deletedSpaces: { ...result.deletedSpaces, ...next.deletedSpaces },
   582→                deletedConversations: { ...result.deletedConversations, ...next.deletedConversations },
   583→                assets: { ...result.assets, ...(next.assets || {}) },
   584→                deletedAssets: { ...result.deletedAssets, ...(next.deletedAssets || {}) },
   585→            };
   586→        }
   587→        return result;
   588→    }

   613→    public async getAsset(id: RemoteId, spaceTag: SpaceTag): Promise<RemoteFilledAsset | RemoteDeletedAsset | null> {
   614→        const data = await this.callGetJson('assets', id);
   615→        if (data === null) return null;
   616→        const asset = convertAssetFromApi(data.Asset, spaceTag);
   617→        if (asset.deleted) return asset;
   618→        if (!asset.encrypted) {
   619→            console.log('getAsset: remote asset does not have the `Encrypted` field set');
   620→            return null;
   621→        }
   622→        // Sorry for the weird juggling below; if I do `return asset`, TypeScript complains that `asset.encrypted`
   623→        // can be undefined, which clearly it cannot because of the `if (!asset.encrypted) return` above.
   624→        // Anyway, this is type safe: it guarantees the return type is cleanly satisfied.
   625→        return { ...asset, encrypted: asset.encrypted };
   626→    }

   644→            queryParams.CreateTimeUntil = String(params.createTimeUntil);
   645→        }
   646→
   647→        let url = '/api/lumo/v1/assets/generated';
   648→        if (Object.keys(queryParams).length > 0) {
   649→            url += `?${new URLSearchParams(queryParams)}`;
   650→        }
   651→
   652→        console.log(`lumo api: http get ${url}`);
   653→        const response = await fetch(url, {
   654→            method: 'GET',
   655→            headers: this.protonHeaders(),
   656→        });
   657→
   658→        if (!response.ok) {
   659→            throw new Error(`HTTP error fetching generated assets: got ${response.status}`);
   660→        }
   661→
   662→        const json = await response.json();
