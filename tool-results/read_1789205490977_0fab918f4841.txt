   134→  interface Events {
   135→    /**
   136→     * Allow, deny, or ask before dispatch. `next()` delegates to allow; missing
   137→     * approval support turns `ask` into denial. Async gates must observe
   138→     * `exec.signal`; the registry rechecks cancellation after they settle but
   139→     * never abandons their promise.
   140→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's calls.
   141→     * @param exec - the pending call (name, parsed arguments, caller agent).
   142→     * @mode waterfall
   143→     */
   144→    'tools/pre-execute'(this: Scoped<ToolRuntime>, exec: ToolExecution, next: () => Promise<PreToolDecision>): Promise<PreToolDecision>
   145→    /**
   146→     * Around-dispatch waterfall for timeout, retry, or metrics. `next()` returns
   147→     * a normalized result; wrappers may change only `exec.signal`, while call
   148→     * identity remains immutable. The registry re-fuses the original caller
   149→     * signal before the body, so replacement cannot detach caller cancellation;
   150→     * wrappers must still restore their signal and reach quiescence.
   151→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's calls.
   152→     * @param exec - the allowed call about to dispatch (name, parsed arguments, caller agent, signal).
   153→     * @mode waterfall
   154→     */
   155→    'tools/execute'(this: Scoped<ToolRuntime>, exec: ToolDispatchExecution, next: () => Promise<ToolExecutionResult>): Promise<ToolExecutionResult>
   156→    /**
   157→     * Accept, replace, enrich, or block a normalized dispatch result. `next()`
   158→     * accepts it unchanged; thrown tools still reach this waterfall as errors. Async
   159→     * listeners must observe `exec.signal`; after they settle, caller
   160→     * cancellation replaces only a successful accepted outcome with the code
   161→     * selected by whether the tool body was invoked.
   162→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's calls.
   163→     * @param exec - the call that just ran (name, parsed arguments, caller agent).
   164→     * @param result - the dispatch outcome a listener may accept, replace, or block.
   165→     * @mode waterfall
   166→     */
   167→    'tools/post-execute'(this: Scoped<ToolRuntime>, exec: ToolExecution, result: Readonly<ToolExecutionResult>, next: () => Promise<PostToolDecision>): Promise<PostToolDecision>
   168→    /**
   169→     * Allow a listener to replace content in the DURABLE LOG COPY of one
   170→     * `run_code` sub-dispatch outcome before the bridge appends its
   171→     * `tool/ptc-dispatch` event. `next()` keeps the
   172→     * content unchanged; a listener may return replacement blocks (e.g. the
   173→     * spill policy's preview + locator for an oversized text result). Only the
   174→     * logged copy is affected — the program already received the complete
   175→     * value, and the model sees neither. A throwing listener is contained:
   176→     * the bridge falls back to logging the original settled content.
   177→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's dispatches.
   178→     * @param dispatch - the parent execution, sub-call identity, and the settled content to log.
   179→     * @mode waterfall
   180→     */
   181→    'tools/ptc-dispatch-log'(this: Scoped<ToolRuntime>, dispatch: PtcDispatchLog, next: () => Promise<ContentBlock[]>): Promise<ContentBlock[]>
   182→    /**
   183→     * Observe the frozen, lossless-JSON final outcome. Listener failures are contained.
   184→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): keyed by `exec.agent`.
   185→     * @param exec - the execution object that traversed the pipeline.
   186→     * @param result - a deep-frozen snapshot of the final returned result.
   187→     * @mode emit
   188→     */
   189→    'tools/result'(this: Scoped<ToolRuntime>, exec: Readonly<ToolExecution>, result: Readonly<ToolExecutionResult>): undefined
   190→    /**
   191→     * A tool was registered or unregistered, or a scoped restriction changed
   192→     * (the available tool set changed — possibly for one scope only). An
   193→     * UNFILTERED registry-subject notification, deliberately not scope-filtered
   194→     * dispatch: a global change concerns every agent's next assembly, so a
   195→     * scoped listener subscribing here sees every change, not just its own
   196→     * scope's.
   197→     * @mode emit
   198→     */
   199→    'tools/change'(): void
   200→  }

   213→/** A registered tool: its schema plus the execution function. */
   214→export interface ToolDefinition extends ToolSchema {
   215→  /** Mandatory canonical output declaration. */
   216→  readonly output: ToolOutputDefinition
   217→  /**
   218→   * Run one accepted call and return only its canonical lossless-JSON value.
   219→   * Async work must observe or forward `exec.signal` and settle only after its
   220→   * owned work reaches quiescence. The registry preserves caller cancellation
   221→   * through around-dispatch signal replacement and does not abandon this
   222→   * promise, but it cannot hard-kill same-process code.
   223→   * @param args - losslessly snapshotted, frozen model arguments.
   224→   * @param exec - execution identity, cancellation signal, and context deferral.
   225→   * @returns the canonical value declared by `output.schema`.
   226→   */
   227→  execute(args: unknown, exec: ToolRunContext): Promise<unknown>
   228→  /**
   229→   * Synchronous last-mile transform for model-facing content. The registry
   230→   * snapshots this callback when execution starts and invokes it exactly once
   231→   * for every normalized outcome, including pipeline failures that bypass
   232→   * `tools/post-execute`, immediately before lossless materialization.
   233→   * Returning `undefined` preserves the content; every other result field
   234→   * remains registry-owned. The callback must be total and must not throw.
   235→   * @param exec - immutable execution identity and arguments.
   236→   * @param result - complete normalized outcome before materialization.
   237→   * @returns replacement content, or `undefined` to preserve it.
   238→   */
   239→  finalizeContent?(exec: Readonly<ToolExecution>, result: Readonly<ToolExecutionResult>): ContentBlock[] | undefined
   240→  /**
   241→   * Cooperative tool-call timeout budget in milliseconds. Omit for no deadline.
   242→   * Enforced by `@deepseek-ai/dsh-tool-call-timeout-policy` (a `tools/execute` wrapper); it
   243→   * is NEVER sent to the model — `schemas()` whitelists only name/description/
   244→   * parameters. Declaring it asserts this tool forwards `exec.signal` to a
   245→   * cooperative implementation that can reach quiescence when the signal aborts.
   246→   */
   247→  timeoutMs?: number
   248→  /**
   249→   * Pure synchronous classifier for overlap with sibling tool calls. Only
   250→   * `true` opts in; omission, exceptions, non-`true` returns, and invalid
   251→   * `defineTool` arguments are exclusive. This metadata is never model-visible.
   252→   *
   253→   * Opted-in executions must not mutate parent-owned state. Shared state must
   254→   * tolerate concurrent dispatch; recorder races are permitted only when they
   255→   * commute or fail closed. See the
   256→   * [parallel-tool-call Agent Note](../../../../.agents/notes/implemented/feature/2026-07-10-parallel-tool-call-execution.md)
   257→   * for the full contract.
   258→   * @param args - parsed arguments; `defineTool` validates before calling.
   259→   * @returns Whether this call may join a parallel group.
   260→   */
   261→  isConcurrencySafe?(args: unknown): boolean
   262→  /**
   263→   * Optional: how to present the PENDING state of one call in a UI, derived from
   264→   * the call's `args` (parsed arguments, `unknown` — the tool validates/narrows
   265→   * its own input). Returns a {@link ToolCallView} (a `card`-tagged render intent),
   266→   * or `undefined` (or omit the method) to fall back to a generic presentation
   267→   * (title = tool name, raw args as input). Pure and side-effect-free: a UI may
   268→   * call it during live streaming AND a session-log replay, so it must depend
   269→   * only on `args`.
   270→   */
   271→  presentCall?(args: unknown): ToolCallView | undefined
   272→  /**
   273→   * Optional: how to present the COMPLETED state, given the same `args` and the
   274→   * durable result projection (`content`, failure state, and optional `meta`). Returns a
   275→   * {@link ToolResultView}, or `undefined` (or omit the method) to keep the
   276→   * pending title and render the raw result content. Pure and side-effect-free
   277→   * for the same replay reason.
   278→   */
   279→  presentResult?(args: unknown, result: ToolResult): ToolResultView | undefined
   280→}

   389→/**

   391→ * accepted a {@link ToolExecution}. {@link deferContext} attaches context to
   392→ * this execution's own result — a composite tool ferries nested-dispatch
   393→ * context back to the outer result, and a leaf tool may mint a fresh
   394→ * plugin-sourced instruction; the loop appends it only after the

   397→export interface ToolRunContext extends ToolExecution {
   398→  /**
   399→   * Defer one context — typically a nested-dispatch context ferried by a
   400→   * composite tool, or a fresh plugin-sourced instruction — until this tool's

   402→   * source and metadata and are emitted in call order.

   410→   * `additionalContexts`, so only an authoritative nested success can

   481→/**
   482→ * Thrown (internally) when the model requests a tool that isn't registered.
   483→ * Extends {@link HarnessError} (`code: 'UNKNOWN_TOOL'`) so an unknown-tool
   484→ * failure is as routable as a tool-thrown one — retry/sandbox/replay code can
   485→ * distinguish it from a tool body's own error.
   486→ */
   487→export class ToolNotFoundError extends HarnessError {
   488→  /**
   489→   * @param toolName - the name the caller asked for.
   490→   * @param reachableFrom - how the model reaches this tool instead, when the
   491→   *   name IS visible and only the presentation denies calling it directly.
   492→   *   Omitted for a name that is registered nowhere.
   493→   */
   494→  constructor(toolName: string, reachableFrom?: string) {
   495→    super(
   496→      reachableFrom === undefined
   497→        ? `unknown tool "${toolName}"`
   498→        : `unknown tool "${toolName}": ${reachableFrom}`,
   499→      'UNKNOWN_TOOL',
   500→    )
   501→    this.name = 'ToolNotFoundError'
   502→  }

   505→/** Thrown when a tool body or post-policy value violates its declared output. */

   507→  /** Schema/value violations in validation order. */
   508→  readonly violations: string[]

   510→  constructor(toolName: string, violations: string[]) {
   511→    super(`tool "${toolName}" returned invalid output: ${violations.join('; ')}`, 'INVALID_TOOL_OUTPUT')
   512→    this.name = 'ToolOutputError'
   513→    this.violations = violations
   514→  }

   517→/** Convert one projector exception into the canonical invalid-output failure. */
   518→function projectionError(toolName: string, projector: 'render' | 'presentationMeta', error: unknown): ToolOutputError {
   519→  return new ToolOutputError(toolName, [`output.${projector} failed: ${errorMessage(error)}`])
   520→}
   521→
   522→/** Snapshot one projector result before later durable-result materialization. */
   523→function snapshotProjection<T>(toolName: string, projector: 'render' | 'presentationMeta', candidate: T): T {
   524→  try {
   525→    const detached = snapshotJsonValue(candidate)
   526→    if (detached === undefined) {
   527→      throw new ToolOutputError(toolName, [`output.${projector} returned non-lossless JSON`])
   528→    }
   529→    return detached
   530→  } catch (error: unknown) {
   531→    if (error instanceof ToolOutputError) throw error
   532→    throw projectionError(toolName, projector, error)
   533→  }
   534→}
   535→
   536→/** Snapshot one body or policy value into the canonical invalid-output failure class. */
   537→function snapshotToolValue(toolName: string, candidate: unknown): JsonValue {
   538→  try {
   539→    const detached = snapshotJsonValue(candidate)
   540→    if (detached === undefined) throw new ToolOutputError(toolName, ['value is not lossless JSON'])
   541→    return detached as JsonValue
   542→  } catch (error: unknown) {
   543→    if (error instanceof ToolOutputError) throw error
   544→    throw new ToolOutputError(toolName, [`value snapshot failed: ${errorMessage(error)}`])
   545→  }
   546→}

   575→/**
   576→ * Pre-dispatch decision. `allow` runs the call; `deny` materializes an error;
   577→ * `ask` runs only after an approval service returns `allowed-once` and otherwise
   578→ * denies. Input rewriting is excluded because arguments are already logged and
   579→ * presented.
   580→ */
   581→export type PreToolDecision =
   582→  | { kind: 'allow' }
   583→  | { kind: 'deny'; reason: string }
   584→  | { kind: 'ask'; reason?: string }

   586→/**
   587→ * Post-dispatch decision: accept, replace one projection, attach context for the
   588→ * next request, or block by turning corrective feedback into an error result.
   589→ */
   590→export type PostToolDecision =
   591→  | { kind: 'accept'; content?: ContentBlock[]; value?: never; additionalContexts?: UserMessage[] }
   592→  | { kind: 'accept'; value: JsonValue; content?: never; additionalContexts?: UserMessage[] }
   593→  | { kind: 'block'; feedback: ContentBlock[]; additionalContexts?: UserMessage[] }
   594→
   595→/**
   596→ * Best-effort human-readable message from an arbitrary thrown value: Error
   597→ * instances use `.message`; non-Error objects with a string `message`
   598→ * property (e.g. `throw { message: 'denied' }`) use it too; everything else
   599→ * is stringified.
   600→ */
   601→function errorMessage(error: unknown): string {
   602→  try {
   603→    if (error instanceof Error) return error.message
   604→    if (typeof error === 'object' && error !== null
   605→      && 'message' in error && typeof error.message === 'string') {
   606→      return error.message
   607→    }
   608→    return String(error)
   609→  } catch {
   610→    // A hostile thrown value can trap `instanceof`, property access, or string
   611→    // coercion. Error normalization is the outermost safety boundary, so its
   612→    // fallback must itself be total.
   613→    return '<unprintable thrown value>'
   614→  }
   615→}
   616→
   617→/** Derive one failure message from policy feedback without changing its rendered blocks. */
   618→function failureMessageFromContent(content: ContentBlock[]): string {
   619→  const text = content
   620→    .map(block => block.type === 'text' ? block.text : `[${block.type} content]`)
   621→    .join('\n')
   622→  return text.length > 0 ? text : 'tool result blocked by post-execute policy'
   623→}

   625→/** Snapshot and freeze one durable tool-result projection or reject lossy data. */
   626→function materializePresentation<T>(candidate: T): T {
   627→  const detached = snapshotJsonValue(candidate)
   628→  if (detached === undefined) {
   629→    throw new TypeError('tool result must be losslessly JSON-serializable')
   630→  }
   631→  return deepFreeze(detached)
   632→}

   639→    return undefined

   706→/** One scope's complete tool-registry contribution. */
   707→class ToolLayer implements ScopeLayer {
   708→  readonly tools: NamedEntries<ToolDefinition>
   709→  readonly restrictions = new AnonymousEntries<CompiledToolRestriction>()
   710→  readonly guards = new AnonymousEntries<ToolGuard>()
   711→  /**
   712→   * Presentation this scope's agent declared for itself, shadowing the
   713→   * deployment default. One cell rather than an entry table: two answers to
   714→   * "which form does the model see" is a contradiction, not a merge.
   715→   */
   716→  mode: ToolPresentationMode | undefined
   717→
   718→  constructor(scope: ScopeKey | undefined) {
   719→    this.tools = new NamedEntries(name => new Error(scope === undefined
   720→      ? `tool "${name}" is already registered (for a per-agent variant, register through that agent's \`agent.ctx\` instead)`
   721→      : `tool "${name}" is already registered in this scope`))
   722→  }
   723→
   724→  /** Whether every contribution table in this aggregate layer is empty. */
   725→  isEmpty(): boolean {
   726→    return this.tools.isEmpty() && this.restrictions.isEmpty() && this.guards.isEmpty()
   727→      && this.mode === undefined
   728→  }
   729→
   730→  /** Whether every compiled restriction in this layer admits a global tool name. */
   731→  admits(name: string): boolean {
   732→    for (const filter of this.restrictions.values()) {
   733→      if ((filter.allow !== undefined && !filter.allow.has(name))
   734→        || (filter.deny !== undefined && filter.deny.has(name))) return false
   735→    }
   736→    return true
   737→  }
   738→
   739→  /** First monotonic denial from this layer's live guard registrations. */
   740→  guardReason(exec: ToolExecution): string | undefined {
   741→    for (const guard of this.guards.values()) {
   742→      const reason = guard(exec)
   743→      if (reason !== undefined) return reason
   744→    }
   745→    return undefined
   746→  }
   747→}

   776→/**
   777→ * Tool registry and execution pipeline. Scoped registrations shadow globals;
   778→ * one visibility resolver feeds presentation, lookup, and dispatch.

   780→export class ToolRuntime extends Service {
   781→  static inject = ['systemPrompt']
   782→
   783→  static Config: z<Config> = z.object({
   784→    mode: z.union(['native', 'ptc', 'both'] as const).default('native'),
   785→    maxParallelSubCalls: z.natural().min(1).default(10),
   786→  })
   787→
   788→  /** Internal staged view consumed by `dsh-agent-loop`'s parallel scheduler. */
   789→  readonly [TOOL_RUNTIME_SCHEDULER]: ToolRuntimeScheduler = {
   790→    prepare: exec => this.prepareScheduledExecution(exec),
   791→    dispatch: exec => this.dispatchScheduledExecution(exec),
   792→    finalize: (exec, result) => this.finalizeScheduledExecution(exec, result),
   793→    finish: (exec, result) => this.finishScheduledExecution(exec, result),
   794→  }
   795→
   796→  /** Context deferred by a running tool body, keyed by its scheduler-owned execution. */
   797→  private deferredContexts = new WeakMap<ToolRunContext, UserMessage[]>()
   798→  /** Executions whose tool body declared the current turn complete. */
   799→  private concludingExecutions = new WeakSet<ToolExecution>()
   800→  /** Original caller cancellation, kept outside the wrapper-mutable execution object. */
   801→  private cancellationStates = new WeakMap<ToolRunContext, ToolCancellationState>()
   802→  /** Definition-owned final content transform snapshotted before policy begins. */
   803→  private contentFinalizers = new WeakMap<ToolRunContext, ToolDefinition['finalizeContent']>()
   804→  private readonly layers = new ScopedLayers(
   805→    scope => new ToolLayer(scope),
   806→    () => { this.ctx.emit('tools/change') },
   807→  )
   808→  /** Presentation for scopes that declare none; {@link presentAs} shadows it per scope. */
   809→  private readonly defaultMode: ToolPresentationMode
   810→  private readonly maxParallelSubCalls: number
   811→  /**
   812→   * Reserved presentation transport, kept outside the filterable registration
   813→   * layers. Built on first need rather than at construction: which agents run
   814→   * a PTC mode is no longer known when the service is constructed, and the
   815→   * transport is stateless beyond its closures over `this`.
   816→   */
   817→  private ptcTransport: ToolDefinition | undefined
   818→
   819→  constructor(ctx: Context, config: Config = {}) {
   820→    super(ctx, 'tools')
   821→    // The schema already defaulted an omitted mode; the ?? narrows the
   822→    // optional-input type for direct (non-Loader) construction in tests.
   823→    this.defaultMode = config.mode ?? 'native'
   824→    this.maxParallelSubCalls = resolveMaxParallelSubCalls(config.maxParallelSubCalls)
   825→    ctx.systemPrompt.tools(context => this.wireSchemas(context.scope))
   826→    if (this.defaultMode !== 'native') {
   827→      ctx.systemPrompt.section(this.collapseSection())
   828→      ctx.systemPrompt.section(this.sdkSection())
   829→    }
   830→  }
   831→
   832→  /**
   833→   * The prompt statement of the `ptc` executor collapse, registered wherever
   834→   * {@link sdkSection} is and rendering empty outside an effective `ptc`.
   835→   *
   836→   * Every tool contributes its own guidance section naming its tool, none of
   837→   * them qualify how that tool is reached, and they all render before the SDK.
   838→   * Without this the model reads a catalog of tools it is told to use and no
   839→   * statement that only `run_code` may be called, so it emits a native call,
   840→   * receives `UNKNOWN_TOOL` for a tool the prompt just declared, and concludes
   841→   * the deployment is inconsistent. Its order places the rule before that
   842→   * guidance rather than after it.
   843→   *
   844→   * `both` renders empty: native calls do execute there, so the rule is false.
   845→   * @returns the section registration.
   846→   */
   847→  private collapseSection(): { name: string; order: number; text: (context: { scope?: ScopeKey }) => string } {
   848→    return {
   849→      name: 'tools:ptc-only',
   850→      order: this.ctx.systemPrompt.getSectionOrder('PTC_ONLY'),
   851→      // The SAME predicate the executor denies by, so the prompt cannot state
   852→      // a rule the registry does not enforce (see `collapses`).
   853→      text: context => this.modeFor(context.scope) === 'ptc' ? PTC_ONLY_INSTRUCTION : '',
   854→    }
   855→  }
   856→
   857→  /**
   858→   * The generated-SDK prompt section, registered globally by a PTC mode
   859→   * deployment and per scope by {@link presentAs}.
   860→   *
   861→   * The body regenerates from the CALLING scope, and renders empty for an
   862→   * agent presenting natively — an agent that opted out under a PTC mode
   863→   * deployment still sees the global registration, and an empty section is
   864→   * dropped from the rendered prompt.
   865→   * @returns the section registration.
   866→   */
   867→  private sdkSection(): { name: string; order: number; text: (context: { scope?: ScopeKey }) => string } {
   868→    return {
   869→      name: 'tools:sdk',
   870→      order: this.ctx.systemPrompt.getSectionOrder('TOOLS_SDK'),
   871→      // Regenerate from the calling scope's visible tools in stable order.
   872→      text: (context) => {
   873→        const mode = this.modeFor(context.scope)
   874→        if (mode === 'native') return ''
   875→        const runtime = this.requireCodeRuntime(mode)
   876→        // Own-property read: a language like `toString`/`constructor` would
   877→        // otherwise resolve an inherited Object.prototype member as a renderer.
   878→        const render = SDK_RENDERERS[runtime.language]
   879→        /* v8 ignore next -- requireCodeRuntime rejects an unknown language before this runs. */
   880→        if (render === undefined) throw new Error(`dsh-tools: no SDK renderer for ${runtime.language}`)
   881→        return render(this.sdkSchemas(context.scope))
   882→      },
   883→    }
   884→  }
   885→
   886→  /**
   887→   * The presentation one scope's agent sees: its own declaration, else the
   888→   * deployment default.
   889→   * @param scope - the calling agent, or undefined for the global view.
   890→   * @returns the resolved presentation mode.
   891→   */
   892→  private modeFor(scope?: ScopeKey): ToolPresentationMode {
   893→    // Nearest scope wins along the chain: a preset's standing declaration
   894→    // covers every agent parented under it, and an agent's own (were one ever
   895→    // declared) would override its preset's. The mode decides what the model
   896→    // SEES, which is exactly the class of fact the chain inherits.
   897→    const layers = this.layers.chainLayers(scope)
   898→    for (let index = layers.length - 1; index >= 0; index -= 1) {
   899→      const mode = layers[index]?.mode
   900→      if (mode !== undefined) return mode
   901→    }
   902→    return this.defaultMode
   903→  }
   904→
   905→  /**
   906→   * The reserved `run_code` transport, built on first need.
   907→   *
   908→   * It never enters the global layer: per-agent restrictions must not remove
   909→   * it, and a scoped registration must not shadow it. The visibility resolver
   910→   * appends it after resolving the filterable global/scoped capability layers,
   911→   * and only for scopes whose mode actually presents it.
   912→   * @returns the shared transport definition.
   913→   */
   914→  private requireCodeTransport(): ToolDefinition {
   915→    this.ptcTransport ??= createRunCodeTool(this, {
   916→      requireRuntime: () => this.requireCodeRuntime(this.defaultMode),
   917→      // The language-aware description/parameters getters read the runtime
   918→      // without demanding one, so a native-default process can still project
   919→      // the transport for an agent that chose code.
   920→      peekRuntime: () => this.ctx.get('codeRuntime'),
   921→      maxParallel: this.maxParallelSubCalls,
   922→      shapeDispatchLog: dispatch => this.shapeDispatchLog(dispatch),
   923→    })
   924→    return this.ptcTransport
   925→  }
   926→
   927→  /**
   928→   * Present the calling scope's tools in `mode` instead of the deployment
   929→   * default. Nearest scope on the chain wins, so a preset's standing
   930→   * declaration covers every agent joined under it.
   931→   *
   932→   * Scoped only, and one declaration per scope: this is how an agent preset
   933→   * composes PTC mode agents beside native ones in the same process, and a
   934→   * process-global override would be the `mode` config field instead.
   935→   * @param mode - the presentation the covered agents' models see.
   936→   * @returns the exact disposer that restores the deployment default.
   937→   */
   938→  presentAs(mode: ToolPresentationMode): () => void {
   939→    const ctx = this.ctx
   940→    if (scopeOf(ctx) === undefined) {
   941→      throw new Error('tools.presentAs() requires a scoped context (agent.ctx): a context-global presentation is the `mode` config field on the tools row')
   942→    }
   943→    const dispose = ctx.effect(function* (this: ToolRuntime) {
   944→      yield this.layers.effect(
   945→        ctx,
   946→        (layer) => {
   947→          if (layer.mode !== undefined) {
   948→            throw new Error(`tools.presentAs("${mode}") conflicts with "${layer.mode}" already declared for this scope; one composition selects one presentation`)
   949→          }
   950→          layer.mode = mode
   951→          return () => { layer.mode = undefined }
   952→        },
   953→        { label: 'tools.presentAs()' },
   954→      )
   955→      // The SDK and collapse sections are per scope for the same reason the
   956→      // mode is. Under a deployment that already defaults to PTC mode this
   957→      // shadows the global registration with an identical body, which costs
   958→      // nothing and keeps one rule instead of a case analysis.
   959→      if (mode !== 'native') {
   960→        yield ctx.systemPrompt.section(this.collapseSection())
   961→        yield ctx.systemPrompt.section(this.sdkSection())
   962→      }
   963→    }.bind(this), 'tools.presentAs()')
   964→    // oxlint-disable-next-line typescript/no-misused-promises -- synchronous composite teardown
   965→    return dispose
   966→  }

   968→  /**
   969→   * Build one scope's wire schemas and names for prompt-order validation.
   970→   * Restrictions do not make known tools invalid, but a mode collapse does.
   971→   */
   972→  private wireSchemas(scope?: ScopeKey): ToolProviderResult {
   973→    const view = this.view(scope)
   974→    const mode = this.modeFor(scope)
   975→    if (mode === 'native') {
   976→      const schemas = [...view.visible.values()].map(definition => this.schemaOf(definition, false))
   977→      return { schemas, knownNames: [...view.knownNames] }
   978→    }
   979→    // Validate the runtime language BEFORE projecting schemas: schemaOf reads
   980→    // run_code's language-aware description/parameters getters, whose own
   981→    // flavor-table guard would otherwise surface first. This keeps the
   982→    // renderer-table rejection the canonical assembly-time error for a
   983→    // language with no SDK renderer.
   984→    this.requireCodeRuntime(mode)
   985→    const schemas = [...view.visible.values()].map(definition => this.schemaOf(definition, false))
   986→    if (mode === 'ptc') {
   987→      return {
   988→        schemas: schemas.filter(schema => schema.name === RUN_CODE_NAME),
   989→        knownNames: [RUN_CODE_NAME],
   990→      }
   991→    }
   992→    return { schemas, knownNames: [...view.knownNames, RUN_CODE_NAME] }
   993→  }
   994→
   995→  /**
   996→   * Resolve the code runtime or throw the actionable misconfiguration error.
   997→   * Read at use time (assembly / run_code execution), NOT via static
   998→   * `inject`: an inject entry would hold `ctx.tools` — and every tool plugin
   999→   * behind it — hostage to a code runtime existing even under `mode:
  1000→   * 'native'`.
  1001→   *
  1002→   * Assembly and `run_code` execution read separately, so the language is not
  1003→   * bound to a request. Harmless while one published backend exists — both
  1004→   * reads return the same flavor — but a reload that swapped in a second
  1005→   * language between them would hand a program written against one SDK to the
  1006→   * other. Binding it is deferred until a second backend ships (the first
  1007→   * point it is testable).
  1008→   */
  1009→  private requireCodeRuntime(mode: ToolPresentationMode): CodeRuntime {
  1010→    const runtime = this.ctx.get('codeRuntime')
  1011→    if (!runtime) {
  1012→      throw new Error(`dsh-tools: mode "${mode}" requires a code runtime — load a ctx.codeRuntime implementation (e.g. @deepseek-ai/dsh-code-runtime-worker-thread) or set tools mode to "native"`)
  1013→    }
  1014→    if (!Object.hasOwn(SDK_RENDERERS, runtime.language)) {
  1015→      const known = Object.keys(SDK_RENDERERS).map(name => JSON.stringify(name)).join(', ')
  1016→      throw new Error(`dsh-tools: no SDK renderer registered for runtime language ${JSON.stringify(runtime.language)} (known: ${known})`)
  1017→    }
  1018→    return runtime
  1019→  }

  1021→  /**
  1022→   * Register globally or in the calling agent scope. Scoped tools shadow
  1023→   * globals; duplicates within one layer and the reserved `run_code` name fail.
  1024→   * @param definition - tool schema, execution, and optional finalization/presentation callbacks.
  1025→   * @returns the exact disposer that unregisters the tool.
  1026→   */
  1027→  register(definition: ToolDefinition): () => void {
  1028→    const name = definition.name
  1029→    const output = (definition as Partial<ToolDefinition>).output
  1030→    if (output === undefined || typeof output !== 'object'
  1031→      || typeof output.render !== 'function'
  1032→      || (output.presentationMeta !== undefined && typeof output.presentationMeta !== 'function')) {
  1033→      throw new TypeError(`tool "${name}" must declare output { schema, render, presentationMeta? }`)
  1034→    }
  1035→    assertSupportedJsonSchema(output.schema)
  1036→    const timeoutMs = definition.timeoutMs
  1037→    if (timeoutMs !== undefined
  1038→      && (!Number.isFinite(timeoutMs) || timeoutMs <= 0)) {
  1039→      throw new TypeError(`tool "${name}" timeoutMs must be a positive finite number`)
  1040→    }
  1041→    // Reserved unconditionally: any agent may select a code mode for itself,
  1042→    // so a name free to take under the deployment default would become a
  1043→    // collision the moment a preset mounted.
  1044→    if (name === RUN_CODE_NAME) {
  1045→      throw new Error(`tool name "${RUN_CODE_NAME}" is reserved for the PTC mode presentation transport and cannot be registered or shadowed`)
  1046→    }
  1047→    return this.layers.effect(
  1048→      this.ctx,
  1049→      layer => layer.tools.insert(name, definition),
  1050→      { label: 'tools.register()' },
  1051→    )
  1052→  }
  1053→
  1054→  /**
  1055→   * Restrict global tools for the calling agent scope. Empty filters, unknown
  1056→   * names, scope-local names, and reserved transport names fail. Restrictions
  1057→   * intersect; scoped registrations remain visible.
  1058→   * @param filter - global-tool mask: `allow` (keep only) and/or `deny` (remove).
  1059→   * @returns the exact disposer that lifts this restriction.
  1060→   */
  1061→  restrict(filter: ToolRestriction): () => void {
  1062→    const scope = scopeOf(this.ctx)
  1063→    if (scope === undefined) {
  1064→      throw new Error('tools.restrict() requires a scoped context (agent.ctx): a context-global restriction would mask every agent — deny the tool for the intended agent instead')
  1065→    }
  1066→    const allow = filter.allow
  1067→    const deny = filter.deny
  1068→    if (allow === undefined && deny === undefined) {
  1069→      throw new Error('tools.restrict({}) is a no-op: pass `allow` and/or `deny` (an empty filter is almost always a materialized-empty-config bug)')
  1070→    }
  1071→    const compiled: CompiledToolRestriction = {
  1072→      ...allow !== undefined ? { allow: new Set(allow) } : {},
  1073→      ...deny !== undefined ? { deny: new Set(deny) } : {},
  1074→    }
  1075→    if ([...allow ?? [], ...deny ?? []].includes(RUN_CODE_NAME)) {
  1076→      throw new Error(`tools.restrict() cannot name reserved PTC mode presentation transport "${RUN_CODE_NAME}"; restrict end-capability tools instead`)
  1077→    }
  1078→    const known = this.view(scope).restrictableNames
  1079→    const unknown = [...allow ?? [], ...deny ?? []].filter(name => !known.has(name))
  1080→    if (unknown.length > 0) {
  1081→      throw new Error(`tools.restrict() names unknown global tool${unknown.length > 1 ? 's' : ''} ${unknown.map(n => `"${n}"`).join(', ')}; known global tools: ${[...known].sort().join(', ') || '(none)'}`)
  1082→    }
  1083→    return this.layers.effect(
  1084→      this.ctx,
  1085→      layer => layer.restrictions.append(compiled),
  1086→      { label: 'tools.restrict()' },
  1087→    )
  1088→  }
  1089→
  1090→  /**
  1091→   * Register a monotonic guard after the extensible `tools/pre-execute`
  1092→   * waterfall. A plain-context guard applies globally; one registered through
  1093→   * `agent.ctx` applies only to that agent. Any matching guard may deny by
  1094→   * returning a reason, while no guard can force-allow a call another guard
  1095→   * denied. The exact effect disposer is returned for ordered ownership and
  1096→   * HMR cleanup.
  1097→   * @param guard - synchronous check; a returned string denies the execution.
  1098→   * @returns the exact disposer that unregisters the guard.
  1099→   */
  1100→  guard(guard: ToolGuard): () => void {
  1101→    return this.layers.effect(
  1102→      this.ctx,
  1103→      layer => layer.guards.append(guard),
  1104→      { label: 'tools.guard()', notify: false },
  1105→    )
  1106→  }
  1107→
  1108→  /** First monotonic denial from the global then the scope chain's guard layers, farthest first. */
  1109→  private guardReason(exec: ToolExecution): string | undefined {
  1110→    const globalReason = this.layers.global.guardReason(exec)
  1111→    if (globalReason !== undefined) return globalReason
  1112→    if (exec.agent === undefined) return undefined
  1113→    for (const layer of this.layers.chainLayers(exec.agent)) {
  1114→      const reason = layer.guardReason(exec)
  1115→      if (reason !== undefined) return reason
  1116→    }
  1117→    return undefined
  1118→  }
  1119→
  1120→  /**
  1121→   * Resolve every registry fact one scope needs in one layer traversal. The
  1122→   * visible map applies restrictions to the INHERITED surface, then the
  1123→   * scope's own registrations and the reserved presentation transport; the
  1124→   * other sets retain the pre-restriction facts needed by restriction and
  1125→   * prompt-order validation.
  1126→   *
  1127→   * A restriction filters what a scope inherits — the global layer and every
  1128→   * ancestor layer on its chain — and never what its OWN layer registers.
  1129→   * That exemption is what a per-child capability filter has to keep intact:
  1130→   * the delegation runtime registers a child's structured-output tool into the
  1131→   * child's own layer, and a filter naming the capabilities the child may use
  1132→   * must not strip the machinery it answers through.
  1133→   *
  1134→   * Reading the exempt set as "the global layer" instead of "not mine" held
  1135→   * only while every model-facing tool sat in the host composition. Once
  1136→   * presets moved them onto the agent plane they became an ANCESTOR
  1137→   * contribution, so a child's filter silently stopped constraining anything
  1138→   * it was given.
  1139→   * @param scope - the viewing scope (the agent), or undefined for the global view.
  1140→   * @returns the complete derived view for that scope.
  1141→   */
  1142→  private view(scope?: ScopeKey): ToolView {
  1143→    // Scope-chain layers, farthest ancestor first, the exact scope last.
  1144→    const layers = this.layers.chainLayers(scope)
  1145→    // Chain-blind on purpose: this is the ONE layer whose registrations the
  1146→    // scope owns rather than inherits, and it is absent until the scope
  1147→    // contributes something.
  1148→    const own = this.layers.peek(scope)
  1149→    // Inherited surface, nearest ancestor last: a nearer scope's same-name
  1150→    // entry shadows a farther one, and the global layer is the farthest.
  1151→    const inherited = new Map<string, ToolDefinition>(this.layers.global.tools.entries())
  1152→    for (const layer of layers) {
  1153→      if (layer === own) continue
  1154→      for (const [name, definition] of layer.tools.entries()) inherited.set(name, definition)
  1155→    }
  1156→    const visible = new Map<string, ToolDefinition>()
  1157→    const knownNames = new Set<string>()
  1158→    const restrictableNames = new Set<string>()
  1159→    for (const [name, definition] of inherited) {
  1160→      knownNames.add(name)
  1161→      restrictableNames.add(name)
  1162→      // Restrictions intersect across the whole chain: any scope on it may
  1163→      // mask an inherited name for everything nested inside it.
  1164→      if (layers.every(layer => layer.admits(name))) visible.set(name, definition)
  1165→    }
  1166→    // The scope's own registrations last, shadowing an inherited name and
  1167→    // outside the filter above.
  1168→    if (own !== undefined) {
  1169→      for (const [name, definition] of own.tools.entries()) {
  1170→        knownNames.add(name)
  1171→        visible.set(name, definition)
  1172→      }
  1173→    }
  1174→    // Presentation infrastructure is resolved last and outside capability
  1175→    // filtering. Registration rejects this reserved name, so the insertion is
  1176→    // an invariant assertion as well as protection against future layer
  1177→    // changes. Per scope: a native agent must not find `run_code` in its
  1178→    // dispatch table because some other agent in the process presents it.
  1179→    if (this.modeFor(scope) !== 'native') {
  1180→      visible.set(RUN_CODE_NAME, this.requireCodeTransport())
  1181→    }
  1182→    return { visible, knownNames, restrictableNames }
  1183→  }

  1185→  /**
  1186→   * Look up a tool as one scope sees it (scoped
  1187→   * shadows global; a restricted-away global reads as absent). Presenters pass
  1188→   * the calling agent so the rendered card matches the definition that
  1189→   * actually executed.
  1190→   * @param name - the tool name as registered.
  1191→   * @param scope - the viewing scope (the agent); omitted = the global view.
  1192→   * @returns the definition the scope resolves, or undefined when none is visible.
  1193→   */
  1194→  get(name: string, scope?: ScopeKey): ToolDefinition | undefined {
  1195→    return this.view(scope).visible.get(name)
  1196→  }

  1198→  /**
  1199→   * Resolve the definition that MAY EXECUTE for a call, applying the mode
  1200→   * collapse at the operation boundary that owns it. The registry view
  1201→   * (`get`) is presentation-agnostic; here a MODEL-DIRECT call under `ptc`
  1202→   * may only name the reserved `run_code` transport, while a nested
  1203→   * sub-dispatch (a `parent` token set — the `run_code` SDK calling a tool
  1204→   * it bound) may call any visible tool. Denial surfaces as `UNKNOWN_TOOL`
  1205→   * through the executor, matching an absent definition.
  1206→   * @param name - the tool name as registered.
  1207→   * @param scope - the viewing scope (the agent); omitted = the global view.
  1208→   * @param nested - whether the call is a transport sub-dispatch, not a model-direct call.
  1209→   * @returns the definition that may run, or undefined when the call must be rejected.
  1210→   */
  1211→  private resolveExecution(name: string, scope: ScopeKey | undefined, nested: boolean): ToolDefinition | undefined {
  1212→    const tool = this.get(name, scope)
  1213→    if (tool === undefined) return undefined
  1214→    if (this.collapses(name, scope, nested)) return undefined
  1215→    return tool
  1216→  }

  1224→  schemas(scope?: ScopeKey): ToolSchema[] {

  1228→  /** Project visible callable tools onto the generated PTC mode SDK contract. */
  1229→  private sdkSchemas(scope?: ScopeKey): ToolSdkSchema[] {
  1230→    return [...this.view(scope).visible.values()]
  1231→      .filter(definition => definition.name !== RUN_CODE_NAME)
  1232→      .map((definition): ToolSdkSchema => {

  1245→  /** Project one definition onto the model-facing schema fields. */
  1246→  private schemaOf(definition: ToolDefinition, detachParameters: boolean): ToolSchema {
  1247→    const { name, description, parameters } = definition
  1248→    const detached = detachParameters ? snapshotJsonValue(parameters) : parameters
  1249→    if (detached === undefined) {
  1250→      throw new Error(`tool "${name}" parameters must be lossless JSON before schema projection`)
  1251→    }
  1252→    return {
  1253→      name,
  1254→      description,
  1255→      parameters: detached,
  1256→    }

  1259→  /**
  1260→   * Classify a pending call through the caller's visible tool definition. Only
  1261→   * an exact `true` is parallel; unknown, hidden, undeclared, invalid, or
  1262→   * throwing classifiers are exclusive.
  1263→   * @param exec - call name, parsed arguments, and optional agent scope.
  1264→   * @returns the fail-closed scheduling mode.
  1265→   */
  1266→  executionMode(exec: ToolExecutionInput): ToolExecutionMode {
  1267→    const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1268→    if (!tool?.isConcurrencySafe) return { kind: 'exclusive' }
  1269→    try {
  1270→      const concurrencySafe: unknown = tool.isConcurrencySafe(exec.arguments)
  1271→      return concurrencySafe === true ? { kind: 'parallel' } : { kind: 'exclusive' }
  1272→    } catch {
  1273→      return { kind: 'exclusive' }
  1274→    }
  1275→  }

  1277→  /**
  1278→   * Run the `tools/ptc-dispatch-log` waterfall over one settled sub-dispatch
  1279→   * and return the content the bridge should log on `tool/ptc-dispatch`.
  1280→   * Contained: when a listener throws, the method logs the original settled
  1281→   * content; that failure must not fail the dispatch or omit the settle event. Private:
  1282→   * the ONE consumer is the `run_code` bridge this registry constructs, which
  1283→   * receives it as a capability parameter (the `requireRuntime` idiom) — the
  1284→   * waterfall, not this invoker, is the public extension point.
  1285→   */
  1286→  private async shapeDispatchLog(dispatch: PtcDispatchLog): Promise<ContentBlock[]> {
  1287→    try {
  1288→      return await this.ctx.waterfall(
  1289→        scopeTarget(this, dispatch.agent), 'tools/ptc-dispatch-log', dispatch,
  1290→        () => Promise.resolve(dispatch.content),
  1291→      )
  1292→    } catch (error: unknown) {
  1293→      this.ctx.logger.warn(`tools: ptc-dispatch-log listener failed for ${dispatch.name}: ${errorMessage(error)}; logging the original settled content`)
  1294→      return dispatch.content
  1295→    }
  1296→  }

  1298→  /**
  1299→   * Whether the `ptc` mode collapse denies a model-direct call: only the
  1300→   * reserved `run_code` transport may be named. Nested sub-dispatches (a
  1301→   * `parent` token set) bypass the collapse. One home for the
  1302→   * security-relevant predicate, shared by {@link resolveExecution} and
  1303→   * {@link createExecution} so the two can never drift apart.
  1304→   *
  1305→   * Resolved through {@link modeFor}, NOT `defaultMode`: an agent given `ptc`
  1306→   * by an agent preset under a native deployment is the composition
  1307→   * `dsh-agent-tool-presentation` exists for, and reading the deployment default would
  1308→   * leave exactly that agent uncollapsed — announcing one surface while
  1309→   * executing another, which is the bypass this collapse closes.
  1310→   * @param name - the tool name as registered.
  1311→   * @param scope - the viewing scope whose effective presentation mode applies.
  1312→   * @param nested - whether the call is a transport sub-dispatch, not a model-direct call.
  1313→   */
  1314→  private collapses(name: string, scope: ScopeKey | undefined, nested: boolean): boolean {
  1315→    return !nested && this.modeFor(scope) === 'ptc' && name !== RUN_CODE_NAME
  1316→  }
  1317→
  1318→  /**
  1319→   * Execute through pre-policy, guards, around-dispatch, post-policy,
  1320→   * definition-owned content finalization, and final notification. Tool and
  1321→   * listener failures resolve as materialized error results; an invisible tool
  1322→   * reports `UNKNOWN_TOOL`. The returned outcome is the same lossless, frozen
  1323→   * snapshot final observers receive. Cancellation
  1324→   * arriving after entry and before final result materialization skips a
  1325→   * not-yet-started body with `ABORTED_BEFORE_DISPATCH` or replaces a
  1326→   * successful started outcome with `ABORTED`; already-started work is still
  1327→   * drained and may retain a tool-owned structured error.
  1328→   * @param exec - the typed same-process call input. The registry assigns its
  1329→   *   correlation token before policy begins.
  1330→   * @returns the materialized final result.
  1331→   */
  1332→  async execute(exec: ToolExecutionInput): Promise<ToolExecutionResult> {
  1333→    return this.prepareExecution(exec, prepared => this.completeScheduledExecution(prepared))
  1334→  }

  1336→  private async completeScheduledExecution(prepared: ScheduledToolPreparation): Promise<ToolExecutionResult> {
  1337→    switch (prepared.kind) {
  1338→      case 'dispatch': {
  1339→        const dispatched = await this.dispatchScheduledExecution(prepared.exec)
  1340→        return dispatched.kind === 'post-result'
  1341→          ? await this.finalizeScheduledExecution(prepared.exec, dispatched.result)
  1342→          : this.finishScheduledExecution(prepared.exec, dispatched.result)
  1343→      }
  1344→      case 'post-result':
  1345→        return await this.finalizeScheduledExecution(prepared.exec, prepared.result)
  1346→      case 'final-result':
  1347→        return this.finishScheduledExecution(prepared.exec, prepared.result)
  1348→      /* v8 ignore next -- closed-union exhaustiveness guard */
  1349→      default:
  1350→        return assertNever(prepared, 'scheduled tool preparation')
  1351→    }
  1352→  }

  1354→  private createExecution(exec: ToolExecutionInput): ScheduledToolPreparation | { kind: 'ready'; exec: MutableToolRunContext } {
  1355→    const deferredContexts: UserMessage[] = []
  1356→    const token = createExecutionToken()
  1357→    const callId = exec.callId
  1358→    const rootCallId = exec.rootCallId ?? callId
  1359→    const name = exec.name
  1360→    const agent = exec.agent
  1361→    const parent = exec.parent
  1362→    const signal = exec.signal
  1363→    // Distinguish a mode-collapsed call (visible in the scope, denied only by
  1364→    // the `ptc` collapse) from a genuinely unknown tool. A collapsed call is
  1365→    // deterministically denied, so it terminates BEFORE the extensible policy
  1366→    // pipeline: pre-execute listeners, approval `ask`, and guards must never
  1367→    // observe — or worse, approve — a call that can only fail. An unknown tool
  1368→    // keeps the historical dispatch-stage `UNKNOWN_TOOL` path so policy
  1369→    // listeners still see every name that reaches the registry.
  1370→    const visible = this.get(name, agent)
  1371→    const collapsed = visible !== undefined && this.collapses(name, agent, parent !== undefined)
  1372→    const concludingExecutions = this.concludingExecutions
  1373→    const base = {
  1374→      token,
  1375→      callId,
  1376→      rootCallId,
  1377→      name,
  1378→      signal,
  1379→      ...agent !== undefined ? { agent } : {},
  1380→      ...parent !== undefined ? { parent } : {},
  1381→      deferContext(context: UserMessage): void {
  1382→        deferredContexts.push(context)
  1383→      },
  1384→      concludeTurn(): void {
  1385→        concludingExecutions.add(this as unknown as ToolExecution)
  1386→      },
  1387→    }
  1388→    // Capture the finalizer BEFORE argument materialization: the
  1389→    // `finalizeContent` contract snapshots the callback when the call starts,
  1390→    // and an arguments getter can replace or clear the registered callback
  1391→    // during `snapshotJsonValue`. The collapse only decides whether the
  1392→    // CAPTURED callback is retained: the pre-dispatch abort path keeps it
  1393→    // (the cancellation contract routes aborted results through it — a getter
  1394→    // that aborts mid-materialization before an invalid-args failure lands in
  1395→    // the same retained path), while the `UNKNOWN_TOOL` denial and the
  1396→    // invalid-args failure of a NON-ABORTED collapsed call drop it (the call
  1397→    // could never execute).
  1398→    const capturedFinalizer = visible?.finalizeContent?.bind(visible)
  1399→    const finalizerFor = (): ToolDefinition['finalizeContent'] | undefined =>
  1400→      collapsed && !signal.aborted ? undefined : capturedFinalizer
  1401→    try {
  1402→      const detached = snapshotJsonValue(exec.arguments)
  1403→      if (detached === undefined) {
  1404→        throw new TypeError('tool execution arguments must be losslessly JSON-serializable')
  1405→      }
  1406→      const execution: MutableToolRunContext = { ...base, arguments: deepFreeze(detached) }
  1407→      this.deferredContexts.set(execution, deferredContexts)
  1408→      this.contentFinalizers.set(execution, finalizerFor())
  1409→      this.cancellationStates.set(execution, {
  1410→        callerSignal: signal,
  1411→        bodyInvoked: false,
  1412→      })
  1413→      if (collapsed) {
  1414→        // The collapse denies the call before the policy pipeline, but a
  1415→        // pre-dispatch abort still keeps the established cancellation
  1416→        // contract: `prepare`'s caller-cancellation check is skipped for
  1417→        // final-results, so honor the abort here instead of surfacing
  1418→        // `UNKNOWN_TOOL` on an already-cancelled call.
  1419→        if (signal.aborted) {
  1420→          return { kind: 'final-result', exec: execution, result: toolAbortedBeforeDispatchResult() }
  1421→        }
  1422→        // The name IS visible here, so the denial carries the route the model
  1423→        // must take instead. Without it the model reads a bare `unknown tool`
  1424→        // for a tool the prompt just declared and concludes the deployment is
  1425→        // broken rather than correcting itself.
  1426→        return {
  1427→          kind: 'final-result',
  1428→          exec: execution,
  1429→          result: toolErrorResult(new ToolNotFoundError(
  1430→            name,
  1431→            `only \`${RUN_CODE_NAME}\` is callable directly — call \`${name}\` from inside a \`${RUN_CODE_NAME}\` program instead`,
  1432→          )),
  1433→        }
  1434→      }
  1435→      return { kind: 'ready', exec: execution }
  1436→    } catch (error: unknown) {
  1437→      const execution: MutableToolRunContext = { ...base, arguments: undefined }
  1438→      this.contentFinalizers.set(execution, finalizerFor())
  1439→      return { kind: 'final-result', exec: execution, result: toolErrorResult(error) }
  1440→    }
  1441→  }

  1443→  /**
  1444→   * Run the ordered pre-execute and monotonic guard stages for the scheduler.
  1445→   * @param input - the caller-supplied execution input.
  1446→   * @returns the prepared execution plus the next scheduler stage.
  1447→   * @internal
  1448→   */
  1449→  private async prepareScheduledExecution(input: ToolExecutionInput): Promise<ScheduledToolPreparation> {
  1450→    return this.prepareExecution(input, prepared => prepared)
  1451→  }

  1453→  private async prepareExecution<T>(
  1454→    input: ToolExecutionInput,
  1455→    next: (prepared: ScheduledToolPreparation) => T | PromiseLike<T>,
  1456→  ): Promise<T> {
  1457→    const created = this.createExecution(input)
  1458→    if (created.kind !== 'ready') return next(created)
  1459→    const exec = created.exec
  1460→    if (this.callerCancelled(exec)) {
  1461→      return next({ kind: 'final-result', exec, result: toolAbortedBeforeDispatchResult() })
  1462→    }
  1463→    try {
  1464→      const carrier = scopeTarget(this, exec.agent)
  1465→      const gate = await this.ctx.waterfall(
  1466→        carrier, 'tools/pre-execute', exec,
  1467→        () => Promise.resolve<PreToolDecision>({ kind: 'allow' }),
  1468→      )
  1469→      const askResolution: ToolAskResolution = gate.kind === 'ask'
  1470→        ? await this.serviceAsk(exec, gate)
  1471→        : { decision: gate, approvalCancelled: false }
  1472→      const { decision } = askResolution
  1473→      if (this.callerCancelled(exec) && askResolution.approvalCancelled) {
  1474→        return await next({ kind: 'post-result', exec, result: toolAbortedBeforeDispatchResult() })
  1475→      }
  1476→      const denialReason = decision.kind === 'allow'
  1477→        ? this.guardReason(exec)
  1478→        : decision.reason
  1479→      if (denialReason !== undefined) {
  1480→        return await next({
  1481→          kind: 'post-result',
  1482→          exec,
  1483→          result: this.materializeFinalResult({
  1484→            content: [{ type: 'text', text: `Error: ${denialReason}` }],
  1485→            isError: true,
  1486→            error: { message: denialReason },
  1487→          }),
  1488→        })
  1489→      }
  1490→      if (this.callerCancelled(exec)) {
  1491→        return await next({ kind: 'post-result', exec, result: toolAbortedBeforeDispatchResult() })
  1492→      }
  1493→      return await next({ kind: 'dispatch', exec })
  1494→    } catch (error: unknown) {
  1495→      return next({ kind: 'final-result', exec, result: toolErrorResult(error) })
  1496→    }
  1497→  }
  1498→
  1499→  /** Whether the original caller signal is currently aborted. */
  1500→  private callerCancelled(exec: ToolRunContext): boolean {
  1501→    const state = this.cancellationStates.get(exec)
  1502→    /* v8 ignore next -- only registry-minted executions reach the staged scheduler methods */
  1503→    if (state === undefined) throw new Error('tool registry scheduler invariant violated: missing cancellation state')
  1504→    return state.callerSignal.aborted
  1505→  }
  1506→
  1507→  /** Canonical cancellation outcome selected by whether the tool body started. */
  1508→  private cancellationResult(exec: ToolRunContext, prior?: ToolExecutionResult): ToolExecutionResult {
  1509→    const state = this.cancellationStates.get(exec)
  1510→    /* v8 ignore next -- only registry-minted executions reach the staged scheduler methods */
  1511→    if (state === undefined) throw new Error('tool registry scheduler invariant violated: missing cancellation state')
  1512→    return state.bodyInvoked
  1513→      ? toolAbortedResult(prior)
  1514→      : toolAbortedBeforeDispatchResult(prior)
  1515→  }
  1516→
  1517→  /**
  1518→   * Dispatch the registered body with the original caller signal fused back
  1519→   * into any around-wrapper replacement. Cancellation never abandons the body:
  1520→   * a started promise reaches quiescence before its outcome becomes `ABORTED`.
  1521→   */
  1522→  private async dispatchToolBody(exec: MutableToolRunContext): Promise<ToolExecutionResult> {
  1523→    const state = this.cancellationStates.get(exec)
  1524→    /* v8 ignore next -- only registry-minted executions reach the staged scheduler methods */
  1525→    if (state === undefined) throw new Error('tool registry scheduler invariant violated: missing cancellation state')
  1526→    const wrapperSignal = exec.signal
  1527→    const fused = fuseToolSignals(state.callerSignal, wrapperSignal)
  1528→    const signal = fused.signal
  1529→
  1530→    if (isAborted(signal)) {
  1531→      fused.dispose()
  1532→      return toolAbortedBeforeDispatchResult()
  1533→    }
  1534→    exec.signal = signal
  1535→    try {
  1536→      const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1537→      if (!tool) throw new ToolNotFoundError(exec.name)
  1538→      state.bodyInvoked = true
  1539→      const returned = await tool.execute(exec.arguments, exec)
  1540→      const result = this.createSuccessResult(exec, tool, returned)
  1541→      return isAborted(signal)
  1542→        ? toolAbortedResult(result)
  1543→        : result
  1544→    } catch (error: unknown) {
  1545→      return toolErrorResult(error)
  1546→    } finally {
  1547→      fused.dispose()
  1548→      exec.signal = wrapperSignal
  1549→    }
  1550→  }
  1551→
  1552→  /**
  1553→   * Run around-dispatch and the tool body. Tool and unknown-tool failures still
  1554→   * receive post-execute; pipeline failures are already final.
  1555→   * @param exec - the prepared execution.
  1556→   * @returns whether the result still needs post-execute.
  1557→   * @internal
  1558→   */
  1559→  private async dispatchScheduledExecution(exec: ToolRunContext): Promise<ScheduledToolDispatch> {
  1560→    try {
  1561→      const mutableExec = exec as MutableToolRunContext
  1562→      const carrier = scopeTarget(this, exec.agent)
  1563→      const result = await this.ctx.waterfall(
  1564→        carrier, 'tools/execute', mutableExec,
  1565→        () => this.dispatchToolBody(mutableExec),
  1566→      )
  1567→      const normalized = this.normalizeDispatchResult(exec, result)
  1568→      const deferredContexts = this.deferredContexts.get(exec)
  1569→      /* v8 ignore next -- dispatch only receives executions minted by this registry's prepare stage */
  1570→      if (deferredContexts === undefined) throw new Error('tool registry scheduler invariant violated: unprepared execution')
  1571→      const resultWithDeferredContexts: ToolExecutionResult = deferredContexts.length === 0
  1572→        ? normalized
  1573→        : this.markCanonical(exec, {
  1574→          ...normalized,
  1575→          additionalContexts: [
  1576→            ...deferredContexts,
  1577→            ...normalized.additionalContexts ?? [],
  1578→          ],
  1579→        })
  1580→      return {
  1581→        kind: 'post-result',
  1582→        result: this.callerCancelled(exec) && !resultWithDeferredContexts.isError
  1583→          ? this.cancellationResult(exec, resultWithDeferredContexts)
  1584→          : resultWithDeferredContexts,
  1585→      }
  1586→    } catch (error: unknown) {
  1587→      return { kind: 'final-result', result: toolErrorResult(error) }
  1588→    }
  1589→  }
  1590→
  1591→  /**
  1592→   * Run ordered post-execute, then apply definition-owned content finalization,
  1593→   * materialize, and notify the final outcome.
  1594→   * @param exec - the prepared execution.
  1595→   * @param result - dispatch/pre result that still needs post-execute.
  1596→   * @returns the materialized final result.
  1597→   * @internal
  1598→   */
  1599→  private async finalizeScheduledExecution(exec: ToolRunContext, result: ToolExecutionResult): Promise<ToolExecutionResult> {
  1600→    try {
  1601→      const postResult = await this.postExecute(exec, result)
  1602→      return this.finishScheduledExecution(
  1603→        exec,
  1604→        this.callerCancelled(exec) && !postResult.isError
  1605→          ? this.cancellationResult(exec, postResult)
  1606→          : postResult,
  1607→      )
  1608→    } catch (error: unknown) {
  1609→      return this.finishScheduledExecution(exec, toolErrorResult(error))
  1610→    }
  1611→  }
  1612→
  1613→  /**
  1614→   * Materialize the candidate, apply definition-owned content finalization,
  1615→   * then materialize and notify the authoritative result.
  1616→   * @param exec - the prepared execution.
  1617→   * @param result - final result.
  1618→   * @returns the materialized final result.
  1619→   * @internal
  1620→   */
  1621→  private finishScheduledExecution(exec: ToolRunContext, result: ToolExecutionResult): ToolExecutionResult {
  1622→    let materializedResult: ToolExecutionResult
  1623→    try {
  1624→      materializedResult = this.materializeFinalResult(result)
  1625→    } catch (error: unknown) {
  1626→      materializedResult = this.materializeFinalResult(toolErrorResult(error))
  1627→    }
  1628→    let finalResult: ToolExecutionResult
  1629→    try {
  1630→      finalResult = this.materializeFinalResult(this.applyFinalContent(exec, materializedResult))
  1631→    } catch (error: unknown) {
  1632→      finalResult = this.materializeFinalResult(toolErrorResult(error))
  1633→    }
  1634→    this.notifyResult(exec, finalResult)
  1635→    return finalResult
  1636→  }
  1637→
  1638→  /** Apply the snapshotted tool-owned content transform without exposing other result fields. */
  1639→  private applyFinalContent(exec: ToolRunContext, result: ToolExecutionResult): ToolExecutionResult {
  1640→    const finalizeContent = this.contentFinalizers.get(exec)
  1641→    if (finalizeContent === undefined) return result
  1642→    const content = finalizeContent(exec, result)
  1643→    return content === undefined ? result : { ...result, content }
  1644→  }
  1645→
  1646→  /** Notify observers without exposing a mutation or error channel into the outcome. */
  1647→  private notifyResult(exec: ToolExecution, result: ToolExecutionResult): void {
  1648→    // Freeze the registry's live object before observers receive its readonly
  1649→    // WeakMap-keyable view.
  1650→    Object.freeze(exec)
  1651→    const { name: toolName, callId } = exec
  1652→    const reportFailure = (error: unknown): void => {
  1653→      this.ctx.logger.warn(`tool "${toolName}" (${callId}): tools/result observer failed: ${errorMessage(error)}`)
  1654→    }
  1655→    const callbacks = this.ctx.events.dispatch('emit', [
  1656→      scopeTarget(this, exec.agent), 'tools/result', exec, result,
  1657→    ])
  1658→    for (const callback of callbacks) {
  1659→      try {
  1660→        const returned: unknown = callback(exec, result)
  1661→        void Promise.resolve(returned).catch(reportFailure)
  1662→      } catch (error: unknown) {
  1663→        reportFailure(error)
  1664→      }
  1665→    }
  1666→  }
  1667→
  1668→  /**
  1669→   * Resolve an `ask` decision to allow/deny through the approval seam. The
  1670→   * seam is consumed opportunistically with `ctx.get('approval')` — a
  1671→   * deployment that composes no ApprovalService keeps the historical degrade
  1672→   * to deny, and an unmount mid-session degrades the same way on the next ask.
  1673→   * An agent-less execution also degrades: without an agent there is no
  1674→   * session to audit to and no UI to route to. Otherwise the outcome maps
  1675→   * one-to-one — `allowed-once` proceeds; the three non-grants deny with
  1676→   * distinct reasons so the model can tell a human "no" from an absent
  1677→   * approval channel.
  1678→   */
  1679→  private async serviceAsk(
  1680→    exec: ToolExecution,
  1681→    ask: Extract<PreToolDecision, { kind: 'ask' }>,
  1682→  ): Promise<ToolAskResolution> {
  1683→    const approval = this.ctx.get('approval')
  1684→    if (approval === undefined) {
  1685→      return {
  1686→        decision: { kind: 'deny', reason: ask.reason ?? `tool "${exec.name}" requires approval (not yet supported)` },
  1687→        approvalCancelled: false,
  1688→      }
  1689→    }
  1690→    if (exec.agent === undefined) {
  1691→      return {
  1692→        decision: { kind: 'deny', reason: `tool "${exec.name}" requires approval, but the call has no agent to route it through` },
  1693→        approvalCancelled: false,
  1694→      }
  1695→    }
  1696→    const outcome = await approval.request({
  1697→      agent: exec.agent,
  1698→      toolName: exec.name,
  1699→      callId: exec.callId,
  1700→      ...ask.reason !== undefined ? { reason: ask.reason } : {},
  1701→      signal: exec.signal,
  1702→    })
  1703→    switch (outcome) {
  1704→      case 'allowed-once': return { decision: { kind: 'allow' }, approvalCancelled: false }
  1705→      case 'rejected': return {
  1706→        decision: { kind: 'deny', reason: `the user rejected tool "${exec.name}"` },
  1707→        approvalCancelled: false,
  1708→      }
  1709→      case 'cancelled': return {
  1710→        decision: { kind: 'deny', reason: `approval for tool "${exec.name}" was cancelled` },
  1711→        approvalCancelled: true,
  1712→      }
  1713→      case 'unavailable': return {
  1714→        decision: { kind: 'deny', reason: `tool "${exec.name}" requires approval, but no approval channel is available` },
  1715→        approvalCancelled: false,
  1716→      }
  1717→      default: return assertNever(outcome, 'ApprovalOutcome')
  1718→    }
  1719→  }
  1720→
  1721→  /**
  1722→   * Run the `tools/post-execute` waterfall over a dispatched `result` and apply
  1723→   * its {@link PostToolDecision}: `accept` keeps the call successful (replacing
  1724→   * `content` when given), `block` turns it into an `isError` whose content is
  1725→   * the corrective `feedback`. Either decision may attach `additionalContexts`,
  1726→   * which are ferried on the returned result for the loop's active-batch FIFO.
  1727→   * Context deferred by the tool body survives an accepted result but is
  1728→   * discarded when the outer call is blocked; a block exposes only context the
  1729→   * blocking decision explicitly supplied.
  1730→   * Runs inside `execute`'s outer try/catch (a throwing listener → isError).
  1731→   */
  1732→  private async postExecute(exec: ToolExecution, result: ToolExecutionResult): Promise<ToolExecutionResult> {
  1733→    const decision = await this.ctx.waterfall(
  1734→      scopeTarget(this, exec.agent), 'tools/post-execute', exec, result,
  1735→      () => Promise.resolve<PostToolDecision>({ kind: 'accept' }),
  1736→    )
  1737→    const decisionContexts = decision.additionalContexts ?? []
  1738→    if (decision.kind === 'block') {
  1739→      const message = failureMessageFromContent(decision.feedback)
  1740→      return this.markCanonical(exec, {
  1741→        content: decision.feedback,
  1742→        isError: true,
  1743→        error: { message },
  1744→        ...decisionContexts.length > 0 ? { additionalContexts: decisionContexts } : {},
  1745→      })
  1746→    }
  1747→    if (Object.hasOwn(decision, 'content') && Object.hasOwn(decision, 'value')) {
  1748→      throw new TypeError('tools/post-execute accept decision cannot replace both value and content')
  1749→    }
  1750→    const additionalContexts = [
  1751→      ...result.additionalContexts ?? [],
  1752→      ...decisionContexts,
  1753→    ]
  1754→    if (Object.hasOwn(decision, 'value')) {
  1755→      if (result.isError) {
  1756→        throw new TypeError('tools/post-execute cannot replace the value of a failed result')
  1757→      }
  1758→      const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1759→      if (tool === undefined) throw new ToolNotFoundError(exec.name)
  1760→      const replaced = this.createSuccessResult(exec, tool, decision.value)
  1761→      return this.markCanonical(exec, {
  1762→        ...replaced,
  1763→        ...additionalContexts.length > 0 ? { additionalContexts } : {},
  1764→      })
  1765→    }
  1766→    return this.markCanonical(exec, {
  1767→      ...result,
  1768→      ...decision.content !== undefined ? { content: decision.content } : {},
  1769→      ...additionalContexts.length > 0 ? { additionalContexts } : {},
  1770→    })
  1771→  }
  1772→
  1773→  /** Registry-normalized results and the exact dispatch that validated each value. */
  1774→  private readonly canonicalResults = new WeakMap<object, ToolExecutionToken>()
  1775→
  1776→  /** Mark one registry-normalized result as canonical only for its owning dispatch. */
  1777→  private markCanonical<T extends ToolExecutionResult>(exec: ToolExecution, result: T): T {
  1778→    this.canonicalResults.set(result, exec.token)
  1779→    return result
  1780→  }

  1782→  /** Snapshot, validate, render, and optionally project one successful body value. */
  1783→  private createSuccessResult(exec: ToolExecution, tool: ToolDefinition, candidate: unknown): ToolExecutionSuccess {
  1784→    const detached = snapshotToolValue(tool.name, candidate)
  1785→    const violations = validateJsonSchemaValue(tool.output.schema, detached, 'value')
  1786→    if (violations.length > 0) throw new ToolOutputError(tool.name, violations)
  1787→    const value = deepFreeze(detached)
  1788→    let rendered: ContentBlock[]
  1789→    try {
  1790→      rendered = tool.output.render(exec.arguments, value)
  1791→    } catch (error: unknown) {
  1792→      throw projectionError(tool.name, 'render', error)
  1793→    }
  1794→    const content = snapshotProjection(tool.name, 'render', rendered)
  1795→    let meta: JsonValue | undefined
  1796→    if (exec.parent === undefined && tool.output.presentationMeta !== undefined) {
  1797→      let projected: JsonValue
  1798→      try {
  1799→        projected = tool.output.presentationMeta(exec.arguments, value)
  1800→      } catch (error: unknown) {
  1801→        throw projectionError(tool.name, 'presentationMeta', error)
  1802→      }
  1803→      meta = snapshotProjection(tool.name, 'presentationMeta', projected)
  1804→    }
  1805→    const concludesTurn = this.concludingExecutions.has(exec)
  1806→    return this.markCanonical(exec, this.materializeFinalResult({
  1807→      isError: false,
  1808→      value,
  1809→      content,
  1810→      ...meta !== undefined ? { meta } : {},
  1811→      ...concludesTurn ? { concludesTurn: true as const } : {},
  1812→    }) as ToolExecutionSuccess)
  1813→  }

  1815→  /** Normalize an around-dispatch wrapper's authored result through the owning output contract. */
  1816→  private normalizeDispatchResult(exec: ToolExecution, result: ToolExecutionResult): ToolExecutionResult {
  1817→    if (this.canonicalResults.get(result) === exec.token) return result
  1818→    if (result.isError) {
  1819→      return this.markCanonical(exec, {
  1820→        isError: true,
  1821→        error: result.error,
  1822→        content: result.content,
  1823→        ...result.meta !== undefined ? { meta: result.meta } : {},
  1824→        ...result.additionalContexts !== undefined ? { additionalContexts: result.additionalContexts } : {},
  1825→      })
  1826→    }
  1827→    const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1828→    if (tool === undefined) throw new ToolNotFoundError(exec.name)
  1829→    const normalized = this.createSuccessResult(exec, tool, result.value)
  1830→    return this.markCanonical(exec, {
  1831→      ...normalized,
  1832→      ...result.additionalContexts !== undefined ? { additionalContexts: result.additionalContexts } : {},
  1833→    })
  1834→  }
  1835→
  1836→  /** Materialize the authoritative commit outcome once, immediately before `tools/result`. */
  1837→  private materializeFinalResult(result: ToolExecutionResult): ToolExecutionResult {
  1838→    const presentation = {
  1839→      content: result.content,
  1840→      ...result.meta !== undefined ? { meta: result.meta } : {},
  1841→      ...result.additionalContexts !== undefined ? { additionalContexts: result.additionalContexts } : {},
  1842→    }
  1843→    if (result.isError) {
  1844→      return materializePresentation({ isError: true as const, error: result.error, ...presentation })
  1845→    }
  1846→    const detached = materializePresentation({
  1847→      isError: false as const,
  1848→      ...presentation,
  1849→      ...result.concludesTurn === true ? { concludesTurn: true as const } : {},
  1850→    })
  1851→    return deepFreeze({ ...detached, value: result.value })
  1852→  }
  1853→}

  1870→/** Read live abort state across an await without treating it as synchronously immutable. */
  1871→function isAborted(signal: AbortSignal): boolean {
  1872→  return signal.aborted

  1875→/**
  1876→ * Fuse caller and wrapper cancellation without nesting `AbortSignal.any`.
  1877→ * Keeping the relay dispatch-scoped also removes listeners when work settles.
  1878→ */
  1879→function fuseToolSignals(caller: AbortSignal, wrapper: AbortSignal): FusedToolSignal {
  1880→  if (caller === wrapper) return { signal: caller, dispose() {} }
  1881→
  1882→  const controller = new AbortController()
  1883→  let listening = false
  1884→  const dispose = (): void => {
  1885→    if (!listening) return
  1886→    listening = false
  1887→    caller.removeEventListener('abort', abortFromCaller)
  1888→    wrapper.removeEventListener('abort', abortFromWrapper)
  1889→  }
  1890→  const abortFrom = (source: AbortSignal): void => {
  1891→    const reason: unknown = source.reason
  1892→    controller.abort(reason)
  1893→    dispose()
  1894→  }
  1895→  const abortFromCaller = (): void => { abortFrom(caller) }
  1896→  const abortFromWrapper = (): void => { abortFrom(wrapper) }
  1897→
  1898→  if (wrapper.aborted) abortFromWrapper()
  1899→  else if (caller.aborted) abortFromCaller()
  1900→  else {
  1901→    listening = true
  1902→    caller.addEventListener('abort', abortFromCaller, { once: true })
  1903→    wrapper.addEventListener('abort', abortFromWrapper, { once: true })
  1904→  }
  1905→  return { signal: controller.signal, dispose }
  1906→}

  1908→/** Canonical result when cancellation supersedes success after body invocation. */
  1909→function toolAbortedResult(prior?: ToolExecutionResult): ToolExecutionResult {
  1910→  const additionalContexts = prior?.additionalContexts ?? []
  1911→  return {
  1912→    content: [{ type: 'text', text: 'Error: tool call aborted' }],
  1913→    isError: true,
  1914→    error: {
  1915→      message: 'tool call aborted',
  1916→      info: { name: 'AbortError', code: TOOL_ABORTED },
  1917→    },
  1918→    ...additionalContexts.length > 0 ? { additionalContexts } : {},
  1919→  }
  1920→}
  1921→
  1922→/** Canonical result when cancellation prevents tool body invocation. */
  1923→function toolAbortedBeforeDispatchResult(prior?: ToolExecutionResult): ToolExecutionResult {

  1933→  }
