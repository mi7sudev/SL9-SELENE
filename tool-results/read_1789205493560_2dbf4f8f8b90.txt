   178→   178→    private async callPostJson(
   179→   179→        resourceName: ResourceName,
   180→   180→        resource: PostableResource,
   181→   181→        path?: string
   182→   182→    ): Promise<RemoteId> {
   183→   183→        const response = await this.sendPostJson(resourceName, resource, path);
   184→   184→        const json = await response.json();
   185→   185→        if (!response.ok) {
   186→   186→            if (response.status === 409) {
   187→   187→                throw new ConflictClientError(`Error during POST ${resourceName}: resource already exists`);
   188→   188→            } else if (response.status === 422 && isLimitReachableResource(resourceName)) {
   189→   189→                // The backend returns 422 when a per-resource limit is reached (e.g.
   190→   190→                // MAX_MESSAGES_PER_CONVERSATION, MAX_ASSETS_PER_SPACE, etc.).
   191→   191→                const code = isProtonApiResponse(json) ? json.Code : undefined;
   192→   192→                const serverMessage =
   193→   193→                    (json && typeof (json as any).Error === 'string' && (json as any).Error) || undefined;
   194→   194→                throw new LimitReachedError(resourceName, { serverMessage, code });
   195→   195→            } else if (response.status >= 400 && response.status < 500) {
   196→   196→                // 4xx don't retry
   197→   197→                throw new ClientError(`Error during POST ${resourceName}: got ${response.status}`);
   198→   198→            }
   199→   199→            throw new Error(`Error during POST ${resourceName}: got ${response.status}`);
   200→   200→        }
   201→   201→        const idExtractor = idExtractorMap[resourceName];
   202→   202→        if (!idExtractor) {
   203→   203→            throw new Error('Unknown resource name');
   204→   204→        }
   205→   205→
   206→   206→        const remoteId = idExtractor(json);
   207→   207→        if (!remoteId) {
   208→   208→            throw new Error("POST: can't find remote ID in response");
   209→   209→        }
   210→   210→        if (!isRemoteId(remoteId)) {
   211→   211→            throw new Error('POST: invalid remote ID in response');
   212→   212→        }
   213→   213→
   214→   214→        return remoteId;
   215→   215→    }

   401→   401→    public async getMasterKey(): Promise<{ eligibility: number; key: Base64 | null }> {
   402→   402→        const data = await this.callListJson('masterkeys');
   403→   403→        const eligibility = data.Eligibility;
   404→   404→
   405→   405→        if (eligibility !== LUMO_ELIGIBILITY.Eligible) {
   406→   406→            return { eligibility, key: null };
   407→   407→        }
   408→   408→
   409→   409→        const keys = convertMasterKeysFromApi(data.MasterKeys);
   410→   410→
   411→   411→        if (keys.length === 0) {
   412→   412→            return { eligibility, key: null };
   413→   413→        }
   414→   414→
   415→   415→        const key = this.findBestKey(keys);
   416→   416→        return { eligibility, key: key.masterKey };

   418→   418→
   419→   419→    private findBestKey(keys: []): undefined;
   420→   420→    private findBestKey(keys: MasterKey[]): MasterKey;
   421→   421→    private findBestKey(keys: MasterKey[]): MasterKey | undefined {
   422→   422→        return keys.reduce((best: MasterKey | undefined, current: MasterKey) => {
   423→   423→            if (!best) return current;
   424→   424→            if (current.isLatest && !best.isLatest) {
   425→   425→                return current;
   426→   426→            } else if (!current.isLatest && best.isLatest) {
   427→   427→                return best;
   428→   428→            } else if (current.version > best.version) {
   429→   429→                return current;
   430→   430→            } else if (current.version < best.version) {
   431→   431→                return best;
   432→   432→            } else {
   433→   433→                // Compare by createdAt if isLatest and version are the same
   434→   434→                const currentCreatedAt = new Date(current.createdAt);
   435→   435→                const bestCreatedAt = new Date(best.createdAt);
   436→   436→                if (currentCreatedAt > bestCreatedAt) {
   437→   437→                    return current;
   438→   438→                } else if (currentCreatedAt < bestCreatedAt) {
   439→   439→                    return best;
   440→   440→                } else {
   441→   441→                    // In case createdAt is the same(!) we need to make a choice,
   442→   442→                    // let's take the lowest id
   443→   443→                    if (current.id < best.id) {
   444→   444→                        return current;
   445→   445→                    } else {
   446→   446→                        return best;
   447→   447→                    }
   448→   448→                }
   449→   449→            }
   450→   450→        }, undefined);
   451→   451→    }

   560→   560→    public async listSpacesPaginate(): Promise<ListSpacesRemote> {
   561→   561→        let result: ListSpacesRemote = {
   562→   562→            spaces: {},
   563→   563→            conversations: {},
   564→   564→            deletedSpaces: {},
   565→   565→            deletedConversations: {},
   566→   566→            assets: {},
   567→   567→            deletedAssets: {},
   568→   568→        };
   569→   569→        let lastTimestamp: Date | undefined;
   570→   570→        while (true) {
   571→   571→            const createTimeUntil = lastTimestamp ? dateToUnixTimestamp(lastTimestamp) : undefined;
   572→   572→            const next = await this.listSpaces({ createTimeUntil });
   573→   573→            const newLastTimestamp = listify({ ...next.spaces, ...next.deletedSpaces })
   574→   574→                .map((s) => new Date(s.createdAt))
   575→   575→                .reduce(oldestDateReducer, lastTimestamp);
   576→   576→            if (!newLastTimestamp || (lastTimestamp && newLastTimestamp >= lastTimestamp)) break;
   577→   577→            lastTimestamp = newLastTimestamp;
   578→   578→            result = {
   579→   579→                spaces: { ...result.spaces, ...next.spaces },
   580→   580→                conversations: { ...result.conversations, ...next.conversations },
   581→   581→                deletedSpaces: { ...result.deletedSpaces, ...next.deletedSpaces },
   582→   582→                deletedConversations: { ...result.deletedConversations, ...next.deletedConversations },
   583→   583→                assets: { ...result.assets, ...(next.assets || {}) },
   584→   584→                deletedAssets: { ...result.deletedAssets, ...(next.deletedAssets || {}) },
   585→   585→            };
   586→   586→        }
   587→   587→        return result;
   588→   588→    }

   614→   614→        const data = await this.callGetJson('assets', id);
   615→   615→        if (data === null) return null;
   616→   616→        const asset = convertAssetFromApi(data.Asset, spaceTag);
   617→   617→        if (asset.deleted) return asset;
   618→   618→        if (!asset.encrypted) {
   619→   619→            console.log('getAsset: remote asset does not have the `Encrypted` field set');
   620→   620→            return null;
   621→   621→        }
   622→   622→        // Sorry for the weird juggling below; if I do `return asset`, TypeScript complains that `asset.encrypted`
   623→   623→        // can be undefined, which clearly it cannot because of the `if (!asset.encrypted) return` above.
   624→   624→        // Anyway, this is type safe: it guarantees the return type is cleanly satisfied.
   625→   625→        return { ...asset, encrypted: asset.encrypted };
