    80→ * Convert Uint8Array to base64 string
    81→ */
    82→function uint8ArrayToBase64(buffer: Uint8Array<ArrayBuffer>): string {
    83→    let binary = '';
    84→    const len = buffer.byteLength;
    85→    for (let i = 0; i < len; i++) {
    86→        binary += String.fromCharCode(buffer[i]!);
    87→    }
    88→    return btoa(binary);
    89→}

    94→function attachmentToWireImage(attachment: Attachment): WireImage {
    95→    // Get data from cache since it's not stored in Redux
    96→    const data = attachmentDataCache.getData(attachment.id);
    97→    if (!data) {
    98→        throw new Error(`Attachment ${attachment.id} has no data in cache`);
    99→    }
   100→    return {
   101→        encrypted: false, // Will be set during encryption phase
   102→        image_id: attachment.id,
   103→        data: uint8ArrayToBase64(data),
   104→    };
   105→}

   118→/**
   119→ * Determine which image attachments should be sent to the backend, keeping only the
   120→ * most recent {@link MAX_IMAGES_PER_REQUEST} images across the whole conversation.
   121→ *
   122→ * Images are ordered chronologically by walking the (post-compaction) chain and
   123→ * applying the same context-filter exclusions used when building turns, so the result
   124→ * matches exactly what would otherwise be sent. Returns the set of image attachment
   125→ * IDs to keep; non-image attachments are unaffected.
   126→ */
   127→function computeKeptImageIds(effectiveChain: Message[], contextFilters: ContextFilter[]): Set<string> {
   128→    const orderedImageIds: string[] = [];
   129→    for (const message of effectiveChain) {
   130→        const filtered = filterMessageAttachments(message.attachments, message.id, contextFilters) ?? [];
   131→        for (const attachment of filtered) {
   132→            if (isImageAttachment(attachment as Attachment)) {
   133→                orderedImageIds.push(attachment.id);
   134→            }
   135→        }
   136→    }
   137→
   138→    if (orderedImageIds.length <= MAX_IMAGES_PER_REQUEST) {
   139→        return new Set(orderedImageIds);
   140→    }
   141→    return new Set(orderedImageIds.slice(orderedImageIds.length - MAX_IMAGES_PER_REQUEST));
   142→}

   144→export function prepareTurns(
   145→    linearChain: Message[],
   146→    personalization: PersonalizationSettings,
   147→    projectInstructions?: string,
   148→    c?: ConversationContext,
   149→    memories?: string,
   150→    agentInstructions?: string,
   151→    includeVisualizationInstructions = false,
   152→    artifactToolMode: ArtifactToolMode = 'off',
   153→    isFromQueryParam = false
   154→): Turn[] {
   155→    // Step 0: Apply any context-compaction boundary. Summarized messages are
   156→    // replaced by a single summary turn and dropped from the chain, so the model
   157→    // only ever receives the compacted view going forward. The full chain remains
   158→    // untouched for display.
   159→    const { summaryTurn, chain: effectiveChain } = collapseCompactedChain(linearChain);
   160→
   161→    // Keep only the most recent images across the conversation; older images beyond
   162→    // the limit are dropped from the request (but remain visible in the UI).
   163→    const keptImageIds = computeKeptImageIds(effectiveChain, c?.contextFilters ?? []);
   164→
   165→    // Step 1: Transform messages to turns by iterating over blocks
   166→    let turns: TurnInProgress[] = [];
   167→
   168→    for (const message of effectiveChain) {
   169→        const blocks = getMessageBlocks(message);
   170→        const contextFilters = c?.contextFilters ?? [];
   171→        const filteredAttachments = filterMessageAttachments(message.attachments, message.id, contextFilters);
   172→        const omittedImageCount = countOmittedImageAttachments(
   173→            message.attachments,
   174→            message.id,
   175→            contextFilters,
   176→            keptImageIds
   177→        );
   178→        const hasAttachmentWork = (filteredAttachments?.length ?? 0) > 0 || omittedImageCount > 0;
   179→
   180→        // Convert each block to appropriate turn(s)
   181→        for (const block of blocks) {
   182→            if (block.type === 'text') {
   183→                // Text block becomes a turn with the message's role
   184→                const textTurn: Turn = {
   185→                    role: message.role,
   186→                    content: sanitizeContent(message.role, block.content),
   187→                };
   188→                turns.push(textTurn);
   189→            } else if (block.type === 'tool_call') {
   190→                // Tool call block becomes a ToolCall turn
   191→                turns.push({
   192→                    role: Role.ToolCall,
   193→                    content: block.content,
   194→                });
   195→            } else if (block.type === 'tool_result') {
   196→                // Tool result block becomes a ToolResult turn
   197→                turns.push({
   198→                    role: Role.ToolResult,
   199→                    content: block.content,
   200→                });
   201→            }
   202→        }
   203→
   204→        // Add attachments to the last turn of this message if it's a user or assistant message.
   205→        // When every image was excluded, still emit placeholder text so vLLM never receives null content.
   206→        if (
   207→            c?.allConversationAttachments &&
   208→            hasAttachmentWork &&
   209→            (message.role === Role.User || message.role === Role.Assistant)
   210→        ) {
   211→            if (turns.length === 0) {
   212→                turns.push({ role: message.role, content: '' });
   213→            }
   214→            const lastTurnIndex = turns.length - 1;
   215→            const lastTurn = turns[lastTurnIndex];
   216→            const attachmentTurns = expandAttachmentsIntoTurns(
   217→                { ...lastTurn, attachments: filteredAttachments ?? [] } as TurnInProgress,
   218→                c.allConversationAttachments,
   219→                keptImageIds,
   220→                omittedImageCount
   221→            );
   222→            // Replace last turn with expanded turns
   223→            turns = [...turns.slice(0, lastTurnIndex), ...attachmentTurns];
   224→        }
   225→    }
   226→
   227→    // Step 2: Insert the final empty assistant turn
   228→    turns.push(EMPTY_ASSISTANT_TURN);
   229→
   230→    // Step 3: (removed) The documentContext string used to be written to turns[i].context, but
   231→    // WireTurn has no `context` field and the backend ignores it. RAG file content is now delivered
   232→    // via proper attachment turns created in Step 1 above (expandAttachmentsIntoTurns), which emit
   233→    // user-role turns with the file content in the `content` field that the API does read.
   234→
   235→    // Step 4a: Inject an artifact-tool nudge as a leading system turn, only when the tool is
   236→    // actually registered for this request (see `resolveArtifactToolMode`) — no tool, no nudge.
   237→    if (artifactToolMode !== 'off') {
   238→        const artifactToolTurn: TurnInProgress = {
   239→            role: Role.System,
   240→            content: artifactToolMode === 'create' ? ARTIFACT_TOOL_CREATE_NUDGE : ARTIFACT_TOOL_REVISE_NUDGE,
   241→        };
   242→        turns = [artifactToolTurn, ...turns];
   243→    }
   244→
   245→    // Step 4b: Add personalization, memories and project instructions to the last user message
   246→    // These are per-request instructions that should apply to the current question
   247→    const personalizationPrompt = formatPersonalization(personalization);
   248→    const instructionParts: string[] = [];
   249→    if (includeVisualizationInstructions) {
   250→        instructionParts.push(VISUALIZATION_INSTRUCTIONS);
   251→    }
   252→    if (agentInstructions) {
   253→        instructionParts.push(`[Agent instructions: ${agentInstructions}]`);
   254→    }
   255→    if (personalizationPrompt) {
   256→        instructionParts.push(`[Personal context: ${personalizationPrompt}]`);
   257→    }
   258→    if (projectInstructions) {
   259→        instructionParts.push(`[Project instructions: ${projectInstructions}]`);
   260→    } else if (memories) {
   261→        instructionParts.push(`[Memories:\n${memories}]`);
   262→    }
   263→
   264→    if (instructionParts.length > 0) {
   265→        const instructionText = instructionParts.join('\n\n');
   266→        const instructionTurn: TurnInProgress = {
   267→            role: Role.System,
   268→            content: instructionText,
   269→        };
   270→        turns = [instructionTurn, ...turns];
   271→    }
   272→
   273→    if (isFromQueryParam) {
   274→        turns = [
   275→            {
   276→                role: Role.System,
   277→                content: QUERY_PARAM_FIRST_INFERENCE_WARNING,
   278→            },
   279→            ...turns,
   280→        ];
   281→    }
   282→
   283→    // Step 4: Remove empty assistant turns
   284→    turns = removeEmptyAssistantTurns(turns);
   285→
   286→    // Step 5: Prepend the compaction summary (if any) as the leading turn so it
   287→    // sits ahead of the preserved messages, enabling prompt-cache coherence.
   288→    if (summaryTurn) {
   289→        turns = [summaryTurn, ...turns];
   290→    }
   291→
   292→    return turns;
   293→}

   325→/**
   326→ * Format a single text attachment for LLM context.
   327→ *
   328→ * Returns `null` when the attachment has no usable content (e.g. project/Drive
   329→ * files whose text lives in the RAG index and whose `markdown` is empty). Emitting
   330→ * an empty `BEGIN/END FILE CONTENTS` block confuses the model into thinking the file
   331→ * is empty, so we omit the block entirely instead — mirroring `flattenAttachmentsForLlm`.
   332→ */
   333→function formatTextAttachmentContent(attachment: Attachment): string | null {
   334→    const beginMarker = '----- BEGIN FILE CONTENTS -----';
   335→    const endMarker = '----- END FILE CONTENTS -----';
   336→
   337→    let content: string | undefined;
   338→    if (attachment.markdown) {
   339→        content = attachment.markdown.trim();
   340→    } else if (attachment.error) {
   341→        content = '[Contents not available: there was an error processing this file]';
   342→    }
   343→
   344→    if (!content) return null;
   345→
   346→    const filename = `Filename: ${attachment.filename}`;
   347→    const header = 'File contents:';
   348→    return [filename, header, beginMarker, content, endMarker].join('\n');
   349→}

   351→/** A single text-file turn produced from an attachment. */
   352→function createTextAttachmentTurn(shallowAttachment: ShallowAttachment, allAttachments: Attachment[]): Turn | null {
   353→    const fullAttachment = allAttachments.find((a) => a.id === shallowAttachment.id);
   354→    if (!fullAttachment) return null;
   355→
   356→    const { imageAttachments, textAttachments } = separateAttachmentsByType([fullAttachment]);
   357→    if (imageAttachments.length > 0 || textAttachments.length === 0) {
   358→        return null; // images are handled separately (grouped into one turn)
   359→    }
   360→
   361→    // Skip when there is no usable content so the model doesn't receive an empty
   362→    // file block (which it interprets as "file is empty").
   363→    const content = formatTextAttachmentContent(fullAttachment);
   364→    if (!content) return null;
   365→
   366→    return { role: Role.User, content };
   367→}

   369→/**
   370→ * Build a single user turn carrying every image attachment, so multimodal models
   371→ * receive all images in one turn instead of one turn per image. Returns null when
   372→ * there are no usable images.
   373→ */
   374→function createGroupedImageTurn(
   375→    shallowAttachments: ShallowAttachment[],
   376→    allAttachments: Attachment[],
   377→    role: Role,
   378→    keptImageIds: Set<string>
   379→): Turn | null {
   380→    const refs: string[] = [];
   381→    const images: WireImage[] = [];
   382→
   383→    for (const shallow of shallowAttachments) {
   384→        const fullAttachment = allAttachments.find((a) => a.id === shallow.id);
   385→        if (!fullAttachment) continue;
   386→
   387→        const { imageAttachments } = separateAttachmentsByType([fullAttachment]);
   388→        if (imageAttachments.length === 0) continue;
   389→
   390→        // Drop images beyond the most-recent-images limit so they aren't sent.
   391→        if (!keptImageIds.has(shallow.id)) continue;
   392→
   393→        const wireImage = tryConvertToWireImage(fullAttachment);
   394→        if (!wireImage) continue;
   395→
   396→        const ref = lumoImageMarker(
   397→            wireImage.image_id,
   398→            role === Role.Assistant ? 'assistant' : 'user',
   399→            shallow.filename
   400→        );
   401→
   402→        refs.push(ref);
   403→        images.push(wireImage);
   404→    }
   405→
   406→    if (images.length === 0) return null;
   407→
   408→    return { role, content: refs.join('\n'), images };
   409→}

   411→/**
   412→ * Filter attachments by removing excluded files based on context filters
   413→ */
   414→function filterMessageAttachments(
   415→    attachments: ShallowAttachment[] | undefined,
   416→    messageId: string,
   417→    contextFilters: ContextFilter[]
   418→): ShallowAttachment[] | undefined {
   419→    if (!attachments || attachments.length === 0) return attachments;
   420→
   421→    const filter = (contextFilters ?? []).find((f) => f.messageId === messageId);
   422→    if (!filter || filter.excludedFiles.length === 0) return attachments;
   423→
   424→    return attachments.filter((att) => !filter.excludedFiles.includes(att.filename));
   425→}

   427→/**
   428→ * Expand a turn's attachments into preceding turns: one turn per text file, plus
   429→ * a single turn carrying all images together (multimodal models expect every
   430→ * image in one turn, not one turn per image). The user's own message turn comes
   431→ * last so the attachments are presented as context to it.
   432→ */
   433→function expandAttachmentsIntoTurns(
   434→    turn: TurnInProgress,
   435→    allAttachments: Attachment[],
   436→    keptImageIds: Set<string>,
   437→    omittedImageCount = 0
   438→): Turn[] {
   439→    const { attachments, ...baseTurn } = turn;
   440→    const shallowAttachments = attachments ?? [];
   441→    const omittedText = formatOmittedImagePlaceholders(omittedImageCount);
   442→
   443→    const textTurns = shallowAttachments
   444→        .map((att) => createTextAttachmentTurn(att, allAttachments))
   445→        .filter((t): t is Turn => t !== null);
   446→
   447→    const imageTurn = createGroupedImageTurn(shallowAttachments, allAttachments, turn.role, keptImageIds);
   448→
   449→    const withOmittedText = (contentParts: (string | undefined)[]) => {
   450→        const content = [...contentParts, omittedText].filter(Boolean).join('\n');
   451→        return content ? { content } : {};
   452→    };
   453→
   454→    if (!imageTurn) {
   455→        return [...textTurns, { ...baseTurn, ...withOmittedText([baseTurn.content]) }];
   456→    }
   457→
   458→    // Merge image markers and bytes into the base turn so that each <lumo-image>
   459→    // marker and its corresponding image_url bytes land in the same Parts content
   460→    // block. The backend zips markers with image_url parts within a single turn;
   461→    // putting them in separate turns breaks UUID assignment on replay.
   462→    const mergedTurn: Turn = {
   463→        ...baseTurn,
   464→        ...withOmittedText([baseTurn.content, imageTurn.content]),
   465→        images: imageTurn.images,
   466→    };
   467→
   468→    return [...textTurns, mergedTurn];
   469→}

   476→    content: Base64,
   477→    u2lParams: { requestKey: AesGcmCryptoKey; requestId: RequestId }
   478→): Promise<string> {

   481→    return decryptString(content, requestKey, ad);

   484→export function postProcessTitle(title: string): string {
   485→    title = title.trim();
   486→    const regex = /"([^"]+)"/;
   487→    const match = title.match(regex);
   488→    if (match) {
   489→        title = match[1]!.trim();
   490→    }
   491→    while (title.endsWith('.')) {
   492→        title = title.substring(0, title.length - 1);
   493→    }
   494→    return title;
   495→}

   497→// todo replace callers to use createReduxCallbacks (lib/lumoApiClientRedux.ts) and remove this function
   498→// todo unused? consider removing
   499→export function _getCallbacks(
   500→    spaceId: SpaceId,
   501→    conversationId: ConversationId,
   502→    assistantMessageId: string,
   503→    u2lParams?: { requestKey: AesGcmCryptoKey; requestId: RequestId }
   504→) {
   505→    let title = '';
   506→    let persistedTitle: string | undefined = undefined;
   507→    let content = '';
   508→
   509→    // todo turn chunkCallback(message, dispatch) into dispatch(chunkCallback(message))
   510→    const chunkCallback = async (m: GenerationResponseMessage, dispatch: LumoDispatch): Promise<{ error?: any }> => {
   511→        switch (m.type) {
   512→            case 'error':
   513→            case 'timeout':
   514→            case 'rejected':
   515→            case 'harmful':
   516→                return {
   517→                    error: createGenerationError(getErrorTypeFromMessage(m.type), conversationId, m),
   518→                };
   519→
   520→            case 'ingesting':
   521→                if (m.target !== 'message' && title) {
   522→                    dispatch(
   523→                        changeConversationTitle({
   524→                            id: conversationId,
   525→                            spaceId,
   526→                            title: postProcessTitle(title),
   527→                            persist: true, // todo review if this flag is still needed
   528→                        })
   529→                    );
   530→                    dispatch(pushConversationRequest({ id: conversationId }));
   531→                    persistedTitle = postProcessTitle(title);
   532→                }
   533→                break;
   534→
   535→            case 'token_data':
   536→                const substring = m.encrypted && u2lParams ? await decryptContent(m.content, u2lParams) : m.content;
   537→                switch (m.target) {
   538→                    case 'title':
   539→                        title += substring;
   540→                        dispatch(
   541→                            changeConversationTitle({
   542→                                id: conversationId,
   543→                                spaceId,
   544→                                title: postProcessTitle(title),
   545→                                persist: false, // todo review if this flag is still needed
   546→                            })
   547→                        );
   548→                        break;
   549→
   550→                    case 'message':
   551→                        content += substring;
   552→                        dispatch(
   553→                            appendChunk({
   554→                                messageId: assistantMessageId,
   555→                                content: substring,
   556→                            })
   557→                        );
   558→                        break;
   559→
   560→                    case 'tool_call':
   561→                        console.log('tool call detected, dispatching setToolCall');
   562→                        dispatch(
   563→                            setToolCall({
   564→                                messageId: assistantMessageId,
   565→                                content: substring,
   566→                            })
   567→                        );
   568→                        break;
   569→
   570→                    case 'tool_result':
   571→                        dispatch(
   572→                            setToolResult({
   573→                                messageId: assistantMessageId,
   574→                                content: substring,
   575→                            })
   576→                        );
   577→                        break;
   578→
   579→                    case 'reasoning':
   580→                        dispatch(
   581→                            appendReasoning({
   582→                                messageId: assistantMessageId,
   583→                                content: substring,
   584→                                sequence: m.count,
   585→                            })
   586→                        );
   587→                        break;
   588→                }
   589→                break;
   590→
   591→            case 'image_data':
   592→                console.log('[IMAGE_DATA] Received in chunkCallback', {
   593→                    image_id: m.image_id,
   594→                    data: m.data ? `${m.data.substring(0, 50)}... (${m.data.length} chars)` : 'none',
   595→                    is_final: m.is_final,
   596→                    seed: m.seed,
   597→                });
   598→
   599→                if (m.image_id && m.data) {
   600→                    const { attachment, data: imageData } = createImageAttachment(m.image_id, m.data, spaceId);
   601→
   602→                    // Store image data in cache instead of Redux
   603→                    attachmentDataCache.setData(attachment.id, imageData);
   604→
   605→                    dispatch(addAttachment(attachment));
   606→                    dispatch(addImageAttachment({ messageId: assistantMessageId, attachment }));
   607→                    dispatch(
   608→                        appendChunk({ messageId: assistantMessageId, content: generateImageMarkdown(m.image_id) })
   609→                    );
   610→                    // Push attachment to server now that it has spaceId
   611→                    dispatch(pushAttachmentRequest({ id: m.image_id }));
   612→                }
   613→                break;
   614→
   615→            case 'usage':
   616→                applyUsageFromStreamMessage(m);
   617→                break;
   618→        }
   619→        return {}; // No error
   620→    };
   621→
   622→    // todo turn finishCallback(status, dispatch) into dispatch(finishCallback(status))
   623→    const finishCallback = async (status: Status, dispatch: LumoDispatch) => {
   624→        dispatch(
   625→            finishMessage({
   626→                messageId: assistantMessageId,
   627→                conversationId,
   628→                spaceId,
   629→                status,
   630→                content: content ?? '',
   631→                role: Role.Assistant,
   632→            })
   633→        );
   634→        dispatch(pushMessageRequest({ id: assistantMessageId }));
   635→
   636→        const finalTitle = postProcessTitle(title);
   637→        if (title) {
   638→            dispatch(
   639→                changeConversationTitle({
   640→                    id: conversationId,
   641→                    spaceId,
   642→                    title: finalTitle,
   643→                    persist: true,
   644→                })
   645→            );
   646→        }
   647→        dispatch(updateConversationStatus({ id: conversationId, status: ConversationStatus.COMPLETED }));
   648→        if (!persistedTitle || finalTitle !== persistedTitle) {
   649→            persistedTitle = finalTitle;
   650→            dispatch(pushConversationRequest({ id: conversationId }));
   651→        }
   652→    };
   653→
   654→    return { chunkCallback, finishCallback };
   655→}
