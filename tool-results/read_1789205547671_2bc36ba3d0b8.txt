    20→/**
    21→ * Get icon name for tool call type.

    23→function getToolCallIcon(toolCall: ToolCallData | ToolCallAnnouncement): IconName {
    24→    if (isNativeToolCallData(toolCall)) {
    25→        if (toolCall.name.startsWith('filesystem__')) {
    26→            return 'Folder';
    27→        }
    28→        if (toolCall.name.startsWith('bash__')) {
    29→            return 'Terminal';
    30→        }
    31→        if (toolCall.name.startsWith('browser__')) {
    32→            return 'Globe';
    33→        }
    34→        if (toolCall.name.startsWith('slack__')) {
    35→            return 'MessageSquare';
    36→        }
    37→        if (toolCall.name.startsWith('confluence__')) {
    38→            return 'FileText';
    39→        }
    40→        return 'Wrench';
    41→    }
    42→
    43→    switch (toolCall.name) {
    44→        case 'web_search':
    45→            return 'Globe';
    46→        case 'weather':
    47→            return 'Cloud';
    48→        case 'stock':
    49→        case 'cryptocurrency':
    50→            return 'TrendingUp';
    51→        case 'describe_image':
    52→        case 'generate_image':
    53→        case 'edit_image':
    54→            return 'Image';
    55→        case 'proton_info':
    56→            return 'Shield';
    57→        default:
    58→            return 'Wrench';
    59→    }
    60→}

    66→function getNativeToolCallLabel(name: string, args: Record<string, unknown>): [string, string] {
    67→    const argStr = (key: string) => (typeof args[key] === 'string' ? (args[key] as string) : '');
    68→
    69→    if (name === 'filesystem__fs_search') {
    70→        const query = argStr('query') || 'files';
    71→        return [`Searching files for "${query}"...`, `Searched files for "${query}" and sent results back`];
    72→    }
    73→    if (name === 'filesystem__fs_read') {
    74→        const path = argStr('path') || 'file';
    75→        return [`Reading ${path}...`, `Read ${path} and sent contents back`];
    76→    }
    77→    if (name === 'filesystem__fs_write') {
    78→        const path = argStr('path') || 'file';
    79→        return [`Writing ${path}...`, `Updated ${path}`];
    80→    }
    81→    if (name === 'filesystem__fs_edit') {
    82→        const path = argStr('path') || 'file';
    83→        return [`Editing ${path}...`, `Edited ${path}`];
    84→    }
    85→    if (name === 'bash__run') {
    86→        const command = argStr('command') || 'command';
    87→        const short = command.length > 60 ? `${command.slice(0, 57)}...` : command;
    88→        return [`Running \`${short}\`...`, `Ran command and sent output back`];
    89→    }
    90→    if (name === 'browser__fetch' || name === 'browser__navigate') {
    91→        const url = argStr('url') || 'page';
    92→        return [`Fetching ${url}...`, `Fetched ${url} and sent contents back`];
    93→    }
    94→
    95→    const shortName = name.includes('__') ? name.split('__').slice(1).join('__') : name;
    96→    return [`Running ${shortName}...`, `Ran ${shortName} and sent results back`];
    97→}

    99→function getToolCallLabel(toolCall: ToolCallData | ToolCallAnnouncement): [string, string] {

   103→    }
   104→
   105→    switch (toolCall.name) {

   132→            return [

   172→function stripMarkdownLine(line: string): string {
   173→    return line
   174→        .replace(/`([^`]+)`/g, '$1')
   175→        .replace(/^#{1,6}\s+/, '')
   176→        .replace(/\*\*([^*]+)\*\*/g, '$1')
   177→        .replace(/__([^_]+)__/g, '$1')
   178→        .replace(/\*([^*\n]+)\*/g, '$1')
   179→        .replace(/_([^_\n]+)_/g, '$1')
   180→        .replace(/\[([^\]]+)\]\([^)]*\)/g, '$1')
   181→        .replace(/^>\s?/, '')
   182→        .replace(/^[-*+]\s+/, '')
   183→        .replace(/^\d+\.\s+/, '')
   184→        .trim();
   185→}

   198→function buildThinkingTraceLines(steps: ThinkingStep[]): string[] {
   199→    const lines: string[] = [];
   200→
   201→    for (const step of steps) {
   202→        if (step.type === 'reasoning') {
   203→            lines.push(...reasoningContentToLines(step.content));
   204→            continue;
   205→        }
   206→
   207→        const [presentLabel, pastLabel] = getToolCallLabel(step.toolCall);
   208→        lines.push(step.isActive ? presentLabel : pastLabel);
   209→    }
   210→
   211→    const activeToolStep = [...steps]
   212→        .reverse()
   213→        .find(
   214→            (step): step is Extract<ThinkingStep, { type: 'tool_call' }> => step.type === 'tool_call' && step.isActive
   215→        );
   216→
   217→    if (activeToolStep) {
   218→        const [presentLabel] = getToolCallLabel(activeToolStep.toolCall);
   219→        if (lines.at(-1) !== presentLabel) {
   220→            lines.push(presentLabel);
   221→        }
   222→    }
   223→
   224→    return lines;
   225→}

   227→function getTraceLineOpacity(index: number, total: number): number {
   228→    if (total <= 1) {
   229→        return 1;
   230→    }
   231→
   232→    const minOpacity = 0.2;
   233→    return minOpacity + (index / (total - 1)) * (1 - minOpacity);

   262→function mergeConsecutiveReasoningSteps(steps: ThinkingStep[]): ThinkingStep[] {
   263→    const merged: ThinkingStep[] = [];
   264→
   265→    for (const step of steps) {
   266→        if (step.type !== 'reasoning') {
   267→            merged.push(step);
   268→            continue;
   269→        }
   270→
   271→        const previous = merged[merged.length - 1];
   272→        if (previous?.type === 'reasoning') {
   273→            merged[merged.length - 1] = {
   274→                ...previous,
   275→                content: previous.content + step.content,
   276→                isActive: step.isActive,
   277→                durationMs: step.durationMs ?? previous.durationMs,
   278→            };
   279→            continue;
   280→        }
   281→
   282→        merged.push({ ...step });
   283→    }
   284→
   285→    return merged;
   286→}

   506→const normalizeWebExtractFailedUrl = (item: unknown): WebExtractFailedUrl | null => {
   507→    if (typeof item === 'string') {
   508→        return { url: item };
   509→    }
   510→
   511→    if (typeof item === 'object' && item !== null && 'url' in item) {
   512→        const failedItem = item as { url: unknown; error?: unknown };
   513→
   514→        if (typeof failedItem.url !== 'string') {
   515→            return null;
   516→        }
   517→
   518→        return {
   519→            url: failedItem.url,
   520→            error: typeof failedItem.error === 'string' ? failedItem.error : undefined,
   521→        };
   522→    }
   523→
   524→    return null;
   525→};
   526→
   527→const parseWebExtractResult = (result: string): WebExtractResult | null => {
   528→    try {
   529→        const parsed = JSON.parse(result) as {
   530→            type?: string;
   531→            results?: WebExtractResultItem[];
   532→            failed_urls?: unknown[];
   533→        };
   534→
   535→        if (parsed.type === 'WebExtract' && Array.isArray(parsed.results)) {
   536→            const failedUrls = Array.isArray(parsed.failed_urls)
   537→                ? parsed.failed_urls
   538→                      .map(normalizeWebExtractFailedUrl)
   539→                      .filter((failedUrl): failedUrl is WebExtractFailedUrl => failedUrl !== null)
   540→                : [];
   541→
   542→            return {
   543→                type: parsed.type,
   544→                results: parsed.results,
   545→                failedUrls,
   546→            };
   547→        }
   548→    } catch {
   549→        // Not valid JSON or not web extract format
   550→    }
   551→    return null;
   552→};
   553→
   554→const parseWebSearchResults = (result: string): WebSearchResults | null => {
   555→    try {
   556→        const parsed = JSON.parse(result);
   557→        if (parsed.results && Array.isArray(parsed.results)) {
   558→            return parsed as WebSearchResults;
   559→        }
   560→    } catch {
   561→        // Not valid JSON or not web search format
   562→    }
   563→    return null;
   564→};

   571→        }
   572→    } catch {
   573→        // Not valid JSON or not image tool format
   574→    }
   575→    return null;

   578→function getToolCallContentLines(toolCall: ToolCallData | ToolCallAnnouncement, result?: string): string[] {
   579→    if (isNativeToolCallData(toolCall)) {
   580→        const args = 'arguments' in toolCall ? toolCall.arguments : {};
   581→        return getNativeToolCallContentLines(toolCall.name, args);
   582→    }
   583→
   584→    const lines: string[] = [];
   585→
   586→    if (toolCall.name === 'web_search') {
   587→        const query = toolCall.arguments?.query;
   588→        if (typeof query === 'string' && query.trim()) {
   589→            lines.push(query.trim());
   590→        }
   591→        if (result) {
   592→            const webSearchResults = parseWebSearchResults(result);
   593→            webSearchResults?.results.forEach((searchResult) => {
   594→                lines.push(searchResult.url);
   595→            });
   596→        }
   597→        return lines;

   914→export const ThinkingPath = ({ steps, message, isThinking, showThinkingTrace, handleLinkClick }: ThinkingPathProps) => {
   915→    const displaySteps = mergeConsecutiveReasoningSteps(steps);
   916→    const [isExpanded, setIsExpanded] = useState(false);
   917→    const isWaitingForSteps = displaySteps.length === 0 && isThinking;
   918→
   919→    const activeHeader = getThinkingPathHeader(displaySteps, message.id, true);
   920→    const completeHeader = getThinkingPathHeader(displaySteps, message.id, false);
   921→    const animatedHeader = useThinkingHeaderAnimation(
   922→        isThinking,
   923→        message.id,
   924→        isWaitingForSteps ? undefined : activeHeader
   925→    );
   926→
   927→    if (displaySteps.length === 0 && !isThinking) {
   928→        return null;
   929→    }
   930→
   931→    const showDone = !isThinking;
   932→    const headerLabel = isThinking ? animatedHeader || activeHeader : completeHeader;
   933→    const traceLines = buildThinkingTraceLines(displaySteps);
   934→    const showTrace = !isExpanded && showThinkingTrace && traceLines.length > 0;
   935→
   936→    return (
   937→        <div className="thinking-path">
   938→            <button
   939→                className={clsx(
   940→                    'thinking-path-header w-full rounded border-none cursor-pointer text-left bg-transparent hover:bg-weak color-weak text-rg lh130',
   941→                    isThinking && 'color-norm'
   942→                )}
   943→                onClick={() => setIsExpanded(!isExpanded)}
   944→                type="button"
   945→                aria-expanded={isExpanded}
   946→                aria-busy={isThinking}
   947→            >
   948→                <span className="inline-flex flex-nowrap items-center min-w-0 text-ellipsis flex-1">
   949→                    <span className={clsx(isThinking && 'text-tabular-nums')}>{headerLabel}</span>
   950→                    {isThinking && <ThinkingProgressDots />}
   951→                </span>
   952→                <LumoIcon
   953→                    name="ChevronDown"
   954→                    width={12}
   955→                    height={12}
   956→                    className={clsx(
   957→                        'thinking-path-header-chevron shrink-0 color-weak',
   958→                        isExpanded && 'thinking-path-header-chevron--expanded',
   959→                        isWaitingForSteps && 'visibility-hidden'
   960→                    )}
   961→                />
   962→            </button>
   963→
   964→            {showTrace && <ThinkingPathTrace lines={traceLines} />}
   965→
   966→            {isExpanded && !isWaitingForSteps && (
   967→                <div className="thinking-path-steps flex flex-nowrap flex-column gap-4 mt-2">
   968→                    {displaySteps.map((step, idx) => {
