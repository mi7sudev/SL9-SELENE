   112→// Error tracking and handling class
   113→export class MockErrorHandler {
   114→    private errorMap: Map<string, number> = new Map();
   115→    private errorCallbacks: Map<string, (() => void)[]> = new Map();
   116→
   117→    // Register an error to be returned for requests matching urlPrefix and method
   118→    // with an optional callback to be called when the error is triggered
   119→    injectError(urlPrefix: string, method: string, statusCode: number, callback?: () => void): void {
   120→        const key = `${method.toUpperCase()}:${urlPrefix}`;
   121→        this.errorMap.set(key, statusCode);
   122→
   123→        // Register the callback if provided
   124→        if (callback) {
   125→            if (!this.errorCallbacks.has(key)) {
   126→                this.errorCallbacks.set(key, []);
   127→            }
   128→            this.errorCallbacks.get(key)!.push(callback);
   129→        }
   130→    }
   131→
   132→    // Clear an injected error
   133→    clearError(urlPrefix: string, method: string): void {
   134→        const key = `${method.toUpperCase()}:${urlPrefix}`;
   135→        const deleted = this.errorMap.delete(key);
   136→        if (!deleted) {
   137→            throw new Error(`cannot clear nonexistent error handler: ${key}`);
   138→        }
   139→        // Also clear any callbacks
   140→        this.errorCallbacks.delete(key);
   141→    }
   142→
   143→    // Clear all registered errors and callbacks
   144→    clearAllErrors(): void {
   145→        this.errorMap.clear();
   146→        this.errorCallbacks.clear();
   147→    }
   148→
   149→    // Check if a request should return an error
   150→    shouldReturnError(url: string, method: string): number | null {
   151→        const requestMethod = method.toUpperCase();
   152→
   153→        // Check each registered error pattern
   154→        for (const [key, statusCode] of this.errorMap.entries()) {
   155→            const [errorMethod, errorUrlPrefix] = key.split(':', 2);
   156→            if (requestMethod === errorMethod && url.startsWith(errorUrlPrefix)) {
   157→                // Trigger any registered callbacks for this error
   158→                this.triggerErrorCallbacks(key);
   159→                return statusCode;
   160→            }
   161→        }
   162→
   163→        return null;
   164→    }
   165→
   166→    private triggerErrorCallbacks(key: string): void {
   167→        const callbacks = this.errorCallbacks.get(key) || [];
   168→        for (const callback of callbacks) {
   169→            callback();
   170→        }
   171→    }
   172→}

   440→// Helper function to check for "undefined" string values in URL or route parameters
   441→function checkForUndefinedParams(
   442→    request: Request,
   443→    params: Record<string, string | readonly string[] | undefined>
   444→): void {
   445→    // Check route parameters
   446→    for (const [key, value] of Object.entries(params)) {
   447→        if (value === 'undefined') {
   448→            const e = new Error(`Route parameter '${key}' has string value 'undefined'`);
   449→            console.error(e);
   450→            throw e;
   451→        }
   452→    }
   453→
   454→    // Check URL query parameters
   455→    const url = new URL(request.url);
   456→    for (const [key, value] of url.searchParams.entries()) {
   457→        if (value === 'undefined') {
   458→            const e = new Error(`URL query parameter '${key}' has string value 'undefined'`);
   459→            console.error(e);
   460→            throw e;
   461→        }
   462→    }
   463→}

   473→// Helper function to parse unix timestamp parameter
   474→function parseUnixTimestamp(paramValue: string | null): string | undefined {
   475→    if (!paramValue) {
   476→        return undefined;
   477→    }
   478→
   479→    // Strict integer parsing - must be all digits, no whitespace, no partial parsing
   480→    if (!/^\d+$/.test(paramValue)) {
   481→        throw new Error(`Invalid timestamp format: ${paramValue}`);
   482→    }
   483→
   484→    const timestamp = parseInt(paramValue, 10);
   485→
   486→    // Additional safety check - ensure the parsed number converts back to the same string
   487→    if (timestamp.toString() !== paramValue) {
   488→        throw new Error(`Invalid timestamp format: ${paramValue}`);
   489→    }
   490→
   491→    return new Date(timestamp * 1000).toISOString();
   492→}
