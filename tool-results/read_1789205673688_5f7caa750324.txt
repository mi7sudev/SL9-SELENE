    87→    console.log('Saga triggered: saveDirtySpace', serializedSpace);
    88→
    89→    // Check if this space is associated with ghost conversations - if so, skip saving to IndexedDB

    94→        return;
    95→    }

    97→    const dbApi: DbApi = yield getContext('dbApi');
    98→    yield call([dbApi, dbApi.updateSpace], serializedSpace, {
    99→        dirty: true,
   100→    });
   101→}

   125→export function* softDeleteSpaceFromRemote({ payload: localId }: { payload: SpaceId }): SagaIterator<any> {
   126→    console.log('Saga triggered: softDeleteSpaceFromRemote', localId);
   127→    const dbApi: DbApi = yield getContext('dbApi');
   128→    const userId: string | undefined = yield select((state: LumoState) => state.user?.value?.ID);
   129→    // IDB cascade first, then Redux cascade. Otherwise a failing IDB write would
   130→    // leave Redux purged while IDB still holds the old data — on reload the state
   131→    // would reappear as a stale tombstone. deleteSpaceCascadeInRedux reads from
   132→    // Redux selectors, so running it after the IDB cascade is still correct.
   133→    yield call([dbApi, dbApi.softDeleteSpaceCascade], localId, { dirty: false });
   134→    try {
   135→        yield call(removeIndexedContentForSpace, localId, userId);
   136→    } catch (error) {
   137→        console.log('softDeleteSpaceFromRemote: failed to remove indexed content, continuing with deletion', error);
   138→    }
   139→    yield call(deleteSpaceCascadeInRedux, localId);
   140→}

   157→export function* serializeSpaceSaga(space: Space): SagaIterator<SerializedSpace> {
   158→    const { id: localId } = space;
   159→    const masterKeyBase64 = yield call(waitForMasterKey, `serializeSpaceSaga ${localId}`);
   160→    const masterKey: AesKwCryptoKey = yield call(base64ToMasterKey, masterKeyBase64);
   161→    const serializedSpace: SerializedSpace | undefined = yield call(serializeSpace, space, masterKey);
   162→    if (!serializedSpace) {
   163→        throw new Error(`serializeRemoteSpace ${localId}: cannot serialize space ${localId} from remote`);
   164→    }
   165→    return cleanSerializedSpace(serializedSpace);

   181→    const type = 'space';
   182→    console.log(`Saga triggered: waitForSpace: ${type} ${localId}`);
   183→    const mapped: Space | undefined = yield select(selectSpaceById(localId));
   184→    if (mapped) {
   185→        console.log(`waitForSpace: requested ${type} ${localId} -> found immediately, returning value`);
   186→        return mapped;
   187→    }
   188→    console.log(`waitForSpace: requested ${type} ${localId} -> not ready, waiting`);
   189→    const { payload: resource }: ReturnType<typeof addSpace> = yield take(
   190→        (a: any) => a.type === addSpace.type && a.payload.id === localId
   191→    );
   192→    console.log(`waitForSpace: requested ${type} ${localId} -> now available, returning value ${resource}`);
   193→    return resource;
   194→}

   309→export function* pushSpace({ payload }: { payload: PushSpaceRequest }): SagaIterator<any> {
   310→    console.log('Saga triggered: pushSpace', payload);
   311→    const type: ResourceType = 'space';
   312→    const { id: localId } = payload;
   313→    const priority = payload.priority || 'urgent';
   314→
   315→    const isGhostChatMode: boolean = yield select((state: LumoState) => state.ghostChat?.isGhostChatMode || false);
   316→    if (isGhostChatMode) {
   317→        yield put(pushSpaceSuccess(payload));
   318→        return;
   319→    }
   320→
   321→    try {
   322→        const dbApi: DbApi = yield getContext('dbApi');
   323→        const idbSpace: SerializedSpace | undefined = yield call([dbApi, dbApi.getSpaceById], localId);
   324→        const remoteId = yield select(selectRemoteIdFromLocal(type, localId));
   325→
   326→        // Deletion case
   327→        if (idbSpace) {
   328→            const { deleted, dirty } = idbSpace;
   329→            // If there is an object in IDB and it has deleted+dirty flags, we push the deletion and exit
   330→            if (deleted) {
   331→                if (dirty) {
   332→                    if (remoteId) {
   333→                        const deleteArgs: [LocalId, RemoteId, Priority] = [localId, remoteId, priority];
   334→                        yield call(callWithRetry, httpDeleteSpace, deleteArgs);
   335→                    }
   336→                    // Clear the dirty flag in IDB, marking the deletion was synced
   337→                    const updated: boolean = yield call(clearDirtyIfUnchanged, idbSpace);
   338→                    // Delete from redux, and if it was done before, it's a noop
   339→                    yield call(deleteSpaceCascadeInRedux, localId);
   340→                    yield put(updated ? pushSpaceSuccess(payload) : pushSpaceNeedsRetry(payload));
   341→                } else {
   342→                    // deletion was already synced, noop
   343→                    yield put(pushSpaceNoop(payload));
   344→                }
   345→                return;
   346→            }
   347→        }
   348→
   349→        const space: Space | undefined = yield select(selectSpaceById(localId));
   350→        if (!space) throw new Error(`cannot find ${type} ${localId} in Redux`);
   351→        const serializedSpace: SerializedSpace = yield call(serializeSpaceSaga, space);
   352→
   353→        if (remoteId) {
   354→            yield call(saveDirtySpace, serializedSpace);
   355→            yield call(callWithRetry, httpPutSpace, [serializedSpace, remoteId, priority]);
   356→            const updated: boolean = yield call(clearDirtyIfUnchanged, serializedSpace);
   357→            yield put(updated ? pushSpaceSuccess(payload) : pushSpaceNeedsRetry(payload));
   358→            return;
   359→        }
   360→
   361→        yield call(saveDirtySpace, serializedSpace);
   362→        const entry = yield call(callWithRetry, httpPostSpace, [serializedSpace, priority]);
   363→        const updated: boolean = yield call(clearDirtyIfUnchanged, serializedSpace);
   364→        yield put(updated ? pushSpaceSuccess({ ...payload, entry }) : pushSpaceNeedsRetry(payload));
   365→    } catch (e) {
   366→        // Retry unless it's a 4xx client error (in which case we expect retrying to fail again)
   367→        console.error(e);
   368→        if (isLimitReachedError(e)) {
   369→            yield put(
   370→                addResourceLimitError({
   371→                    resource: 'spaces',
   372→                    limit: MAX_SPACES_PER_USER,
   373→                    serverMessage: e.serverMessage,
   374→                })
   375→            );
   376→            yield put(pushSpaceFailure({ ...payload, error: `${e}` }));
   377→        } else if (isClientError(e)) {
   378→            yield put(pushSpaceFailure({ ...payload, error: `${e}` }));
   379→        } else {
   380→            yield put(pushSpaceNeedsRetry(payload));
   381→        }
   382→    }
   383→}

   393→export function* waitForSpaceRemoteId(localId: SpaceId): SagaIterator<SpaceId | undefined> {
   394→    console.log('Saga triggered: waitForSpaceRemoteId', localId);
   395→    const type = 'space';
   396→    const existingRemoteId: RemoteId | undefined = yield select((s: LumoState) => s.idmap.local2remote[type][localId]);
   397→    if (existingRemoteId) {
   398→        return existingRemoteId;
   399→    }
   400→
   401→    // Check if that space is local-only; if it was never pushed, it's useless to query the server
   402→    const localSpace: Space | undefined = yield select(selectSpaceById(localId));
   403→    if (localSpace) {
   404→        console.log('waitForSpaceRemoteId: space was created on this device and is not yet pushed to the server');
   405→        return undefined;
   406→    }
   407→
   408→    // Trigger a general listing of spaces, which contains local-remote id mapping for each space
   409→    yield put(pullSpacesRequest());
   410→
   411→    // Wait until the pullSpacesRequest is done...
   412→    const result: ReturnType<typeof pullSpacesSuccess | typeof pullSpacesFailure> = yield take([
   413→        pullSpacesSuccess.type,
   414→        pullSpacesFailure.type,
   415→    ]);
   416→
   417→    // Nothing we can do if pullSpacesRequest failed
   418→    if (result.type === pullSpacesFailure.type) {
   419→        console.error(`waitForSpaceRemoteId: Failed to pull spaces list for ${localId}`);
   420→        yield put(pullSpaceFailure(localId));
   421→        return undefined;
   422→    }
   423→
   424→    // PullSpaces was successful; check if we have the mapping now
   425→    const remoteId: RemoteId | undefined = yield select((s: LumoState) => s.idmap.local2remote[type][localId]);
   426→    if (!remoteId) {
   427→        console.error(`waitForSpaceRemoteId: Remote ID not found for ${localId} even after pulling spaces`);
   428→        yield put(pullSpaceFailure(localId));
   429→        return undefined;
   430→    }
   431→
   432→    // We now have the mapping! return it
   433→    return remoteId;
   434→}

   436→export function* pullSpace({ payload }: { payload: PullSpaceRequest }): SagaIterator<void> {
   437→    const { id: localId } = payload;
   438→    console.log('Saga triggered: pullSpace', localId);
   439→    try {
   440→        const remoteId: SpaceId | undefined = yield call(waitForSpaceRemoteId, localId);
   441→        if (!remoteId) {
   442→            console.error(`pullSpace: Remote ID not found for ${localId}`);
   443→            yield put(pullSpaceFailure(localId));
   444→            return;
   445→        }
   446→
   447→        const lumoApi: LumoApi = yield getContext('lumoApi');
   448→        const result: GetSpaceRemote | null = yield call([lumoApi, lumoApi.getSpace], remoteId);
   449→        if (!result) {
   450→            console.error(`pullSpace: Space ${localId} not found on server`);
   451→            yield put(pullSpaceFailure(localId));
   452→            return;
   453→        }
   454→        yield put(pullSpaceSuccess(result));
   455→    } catch (e) {
   456→        console.error('Error pulling space:', e);
   457→        yield put(pullSpaceFailure(localId));
   458→    }
   459→}

   461→export function* pullSpaces(): SagaIterator<void> {
   462→    console.log('Saga triggered: pullSpaces');
   463→    try {
   464→        const lumoApi: LumoApi = yield getContext('lumoApi');
   465→        let lastTimestamp: number | undefined;
   466→
   467→        while (true) {
   468→            const listSpacesArgs = lastTimestamp ? { createTimeUntil: lastTimestamp } : undefined;
   469→            const pageResult: ListSpacesRemote = yield call([lumoApi, lumoApi.listSpaces], listSpacesArgs);
   470→            yield put(pullSpacesPageResponse(pageResult));
   471→
   472→            const allSpaces = [...listify(pageResult.spaces), ...listify(pageResult.deletedSpaces)];
   473→            if (allSpaces.length === 0) break;
   474→
   475→            const newLastTimestamp = allSpaces
   476→                .map((s) => s.createdAt)
   477→                .map((iso: string): number => isoToUnixTimestamp(iso))
   478→                .reduce((oldest, current) => {
   479→                    if (!oldest) return current;
   480→                    return current <= oldest ? current : oldest;
   481→                }, lastTimestamp);
   482→
   483→            if (!newLastTimestamp || (lastTimestamp && newLastTimestamp >= lastTimestamp)) break;
   484→
   485→            lastTimestamp = newLastTimestamp;
   486→            yield delay(1000);
   487→        }
   488→
   489→        yield put(pullSpacesSuccess());
   490→    } catch (e) {
   491→        console.error('Error pulling spaces:', e);
   492→        yield put(pullSpacesFailure());
   493→    }
   494→}

   496→export function* processPullSpacesGeneric(payload: PullSpacesRemote): SagaIterator<any> {
   497→    const { conversations, deletedConversations, deletedSpaces, spaces, assets, deletedAssets, requestFull } = payload;
   498→    for (const space of listify(deletedSpaces)) {
   499→        const { wrappedSpaceKey, ...deletedSpace } = space;
   500→        yield put(locallyDeleteSpaceFromRemoteRequest(deletedSpace.id));
   501→    }
   502→    for (const space of listify(spaces)) {
   503→        yield put(locallyRefreshSpaceFromRemoteRequest(space));
   504→    }
   505→    for (const conversation of listify(deletedConversations)) {
   506→        yield put(locallyDeleteConversationFromRemoteRequest(conversation.id));
   507→    }
   508→    for (const conversation of listify(conversations)) {
   509→        yield put(locallyRefreshConversationFromRemoteRequest(conversation));
   510→    }
   511→    for (const asset of listify(deletedAssets)) {
   512→        yield put(locallyDeleteAttachmentFromRemoteRequest(asset.id));
   513→    }
   514→    for (const asset of listify(assets)) {
   515→        yield put(locallyRefreshShallowAttachmentFromRemoteRequest({ ...asset, requestFull }));
   516→    }

   551→export function* refreshSpaceFromRemote({
   552→    payload: encryptedRemoteSpace,
   553→}: {
   554→    payload: RemoteSpace;
   555→}): SagaIterator<any> {
   556→    console.log('Saga triggered: refreshSpaceFromRemote', encryptedRemoteSpace.id);
   557→    const type = 'space';
   558→    const dbApi: DbApi = yield getContext('dbApi');
   559→    const { id: localId, remoteId } = encryptedRemoteSpace;
   560→
   561→    const masterKeyBase64 = yield call(waitForMasterKey, `refreshSpaceFromRemote ${localId}`);
   562→    const masterKey: AesKwCryptoKey = yield call(base64ToMasterKey, masterKeyBase64);
   563→    const deserializedRemoteSpace: Space | null | undefined = yield call(
   564→        deserializeSpace,
   565→        encryptedRemoteSpace,
   566→        masterKey
   567→    );
   568→    if (!deserializedRemoteSpace) {
   569→        console.error(`refreshSpaceFromRemote ${localId}: cannot deserialize space`);
   570→        return;
   571→    }
   572→    const remoteSpace = cleanSpace(deserializedRemoteSpace);
   573→
   574→    const localSpace: Space | undefined = yield select(selectSpaceById(localId));
   575→    const encryptedIdbSpace: SerializedSpace | undefined = yield call([dbApi, dbApi.getSpaceById], localId);
   576→
   577→    if (localSpace) {
   578→        const isDirty = encryptedIdbSpace?.dirty || false;
   579→
   580→        if (isDirty) {
   581→            console.log(`refreshSpaceFromRemote ${localId}: Local has dirty changes, keeping local`);
   582→            return;
   583→        }
   584→
   585→        console.log(`refreshSpaceFromRemote ${localId}: Updating from remote`);
   586→        yield put(addSpace(remoteSpace));
   587→        yield call([dbApi, dbApi.updateSpace], encryptedRemoteSpace, { dirty: false });
   588→        return;
   589→    }
   590→
   591→    if (encryptedIdbSpace) {
   592→        const idbSpace: Space | null | undefined = yield call(deserializeSpace, encryptedIdbSpace, masterKey);
   593→        const isDirty = encryptedIdbSpace.dirty || false;
   594→
   595→        if (isDirty && idbSpace) {
   596→            console.log(`refreshSpaceFromRemote ${localId}: IDB has dirty changes, keeping IDB`);
   597→            const cleanIdb = cleanSpace(idbSpace);
   598→            yield put(addSpace(cleanIdb));
   599→            yield put(addIdMapEntry({ type, localId, remoteId, saveToIdb: false }));
   600→            return;
   601→        }
   602→
   603→        console.log(`refreshSpaceFromRemote ${localId}: Updating from remote`);
   604→        yield put(addSpace(remoteSpace));
   605→        yield put(addIdMapEntry({ type, localId, remoteId, saveToIdb: true }));
   606→        yield call([dbApi, dbApi.updateSpace], encryptedRemoteSpace, { dirty: false });
   607→        return;
   608→    }
   609→
   610→    console.log(`refreshSpaceFromRemote ${localId}: New space, inserting from remote`);
   611→    yield put(addSpace(remoteSpace));
   612→    yield put(addIdMapEntry({ type, localId, remoteId, saveToIdb: true }));
   613→    yield call([dbApi, dbApi.updateSpace], encryptedRemoteSpace, { dirty: false });
   614→}

   616→export function* reconcileProjectSearchIndex(): SagaIterator {
   617→    try {
   618→        const userId: string | undefined = yield select((state: LumoState) => state.user?.value?.ID);
   619→        if (!userId) {
   620→            return;
   621→        }
   622→
   623→        const spaces: Record<string, Space> = yield select((state: LumoState) => state.spaces);
   624→        const attachments: AttachmentMap = yield select((state: LumoState) => state.attachments);
   625→        const indexedDriveFolders: { spaceId?: string }[] =
   626→            (yield select((state: LumoState) => state.lumoUserSettings.indexedDriveFolders)) || [];
   627→
   628→        for (const space of Object.values(spaces)) {
   629→            if (!space.isProject || space.linkedDriveFolder) {
   630→                continue;
   631→            }
   632→
   633→            if (indexedDriveFolders.some((folder) => folder.spaceId === space.id)) {
   634→                yield call(removeIndexedContentForSpace, space.id, userId, { documentScope: 'drive-only' });
   635→            }
   636→        }
   637→
   638→        const snapshots = buildProjectKnowledgeSnapshots(spaces, attachments);
   639→        if (snapshots.length === 0) {
   640→            return;
   641→        }
   642→
   643→        const searchService = SearchService.get(userId);
   644→        const removed: number = yield call([searchService, searchService.reconcileStaleProjectDocuments], snapshots);
   645→        if (removed > 0) {
   646→            console.log('[reconcileProjectSearchIndex] Removed stale project documents:', removed);
   647→        }
   648→    } catch (error) {
   649→        console.error('[reconcileProjectSearchIndex] Failed:', error);
   650→    }
   651→}
